            TlsListener::EventMask TlsListener::getSessionEventMask(Session& session) noexcept
            {
                if (m_pFeatureStatusManager == nullptr || !m_pFeatureStatusManager->getFeatureStatus()) {
                    return 0;
                }

                if (m_httpsPortsViewer == nullptr ||
                    !m_httpsPortsViewer->isHttpsPort(session.getKey()->getDstPort())) {
                    return 0;
                }

                DPPolicyItem* policy = DPSharedDoublePolicyArray::getGlobalSharedPolicyArray()->getActivePolicy(session.getPolicyId());
                if (policy == nullptr || !policy->profileExists[IDS_PROFILE_TLS_ENFORCEMENT]) {
                    return 0;
                }

                return 0b00111111; // all 6 events (ClientHello, Anomaly, TCPReset, SessionEnd, TLSRecord, TerminalPayload)
            }

            TlsListener::TLSFeatureVerdict TlsListener::onClientHello(Session& session, DPFrame& packet, const ClientHelloData& data, const TLSData& tlsData) noexcept
            {
                if (m_detector != nullptr) {
                    Violation violation = m_detector->onParsedClientHello(session, packet, data, tlsData);
                    if (ViolationAPI::hasAny(violation)) {
                        record(packet, violation);
                        const TLSFlow::TLSViolationMask verdictMask =
                            ViolationAPI::toTLSViolationMask(violation);
                        if (m_detector->getPacketAction() == Action::BlockAndReport) {
                            return TLSFeatureVerdict{TLSFeatureVerdict::Dispatch::Stop, verdictMask};
                        }
                        return TLSFeatureVerdict{TLSFeatureVerdict::Dispatch::Continue, verdictMask};
                    }
                }
                return TLSFeatureVerdict{};
            }

            void TlsListener::record(DPFrame& packet, Violation violation) noexcept
            {
                // Priority order matches FailureType enum order (0=Malformed, 1=Version, etc.)
                // Report only the first (highest priority) violation found
                int firstSetBit = __builtin_ffs(violation);
                if (firstSetBit > 0) {
                    FailureType failureType = static_cast<FailureType>(firstSetBit - 1);

                    // Record statistics first
                    if (m_statisticsRecorder != nullptr) {
                        m_statisticsRecorder->recordPacket(TLSEnforcement::Statistics::Common::Key(packet.getPolicyId(), failureType), 1, packet.getFrameTotalLen());
                    }

                    // Send reporting sample
                    if (m_reportingManager != nullptr) {
                        m_reportingManager->handleReport(&packet, failureType, 1);
                    }
                }
            }

                                            handshakeAgingTimeDetector{sessionParamsViewer, featureStatusManager, handshakeAgingTimeManager, diagnosticsCounters},
                                            wrongOrderDetector{sessionParamsViewer, featureStatusManager, maxEarlyDataManager, diagnosticsCounters},
                                            ruleBasedDetector{sessionParamsViewer, featureStatusManager, diagnosticsCounters},

            /// Event Handlers

            Violation Manager::handleEvent(Detector& detectionStrategy, Session& session, DPFrame* packet, const TLSData &tlsData) noexcept
            {
                Violation violation = detectionStrategy.detect(session, tlsData);
                if (!ViolationAPI::hasAny(violation) || !m_mitigator) {
                    return violation;
                }

                const Violation newViolation = ViolationAPI::filterAlreadySeen(
                    violation,
                    tlsData.terminalViolationMask);

                if (ViolationAPI::hasAny(newViolation)) {
                    if (packet != nullptr) {
                        m_mitigator->mitigate(session, *packet, newViolation);
                    } else {
                        m_mitigator->mitigate(session, newViolation);
                    }
                    return violation;
                }

                if (packet != nullptr &&
                    m_sessionParamsViewer != nullptr &&
                    m_sessionParamsViewer->getPacketAction() == Action::BlockAndReport) {
                    Violation packetActionOnly = NoViolation;
                    ViolationAPI::set(packetActionOnly, FailureType::MalformedSession);
                    m_mitigator->mitigate(session, *packet, packetActionOnly);
                }

                return violation;
            }

            Violation Manager::onParsedClientHello(Session &session, DPFrame &packet, const ClientHelloData &clientHelloData, const TLSData &tlsData) noexcept
            {
                m_detectorsPackPimpl->ruleBasedDetector.setCurrentClientHelloData(&clientHelloData);
                Detector& detector = m_detectorsPackPimpl->ruleBasedDetector;
                return handleEvent(detector, session, &packet, tlsData);
            }

            Violation Manager::onParsingError(Session &session, DPFrame &packet, ParsingError parsingError, const TLSData &tlsData) noexcept
            {
                ParsingErrorDetector& detector = m_detectorsPackPimpl->parsingErrorDetector;
                detector.observeParsingError(parsingError);
                detector.setParsingError(parsingError);
                Violation violation = handleEvent(detector, session, &packet, tlsData);
                detector.clearParsingError();
                return violation;
            }

            Violation Manager::onTLSRecord(Session &session, DPFrame &packet, const TLSFlow::PacketFlow::Engine::TLSRecordContext& context, const TLSData &tlsData) noexcept
            {
                if (context.record.isAlert()) {
                    if (!context.record.readableAlert || !context.alertTerminating) {
                        return NoViolation;
                    }

                    Detector& detector = m_detectorsPackPimpl->alertDetector;
                    (void)handleEvent(detector, session, nullptr, tlsData);
                    return NoViolation;
                }

                WrongOrderDetector& detector = m_detectorsPackPimpl->wrongOrderDetector;
                detector.setLastRecordSeen(&context.record);
                detector.setCurrentClientHelloSupportedVersionsMask(context.currentClientHelloSupportedVersionsMask);
                detector.setEarlyDataOffered(context.earlyDataOffered);
                return handleEvent(detector, session, &packet, tlsData);
            }

            Violation Manager::onTerminalPayload(Session &session, DPFrame &packet, const TLSData &tlsData) noexcept
            {
                Detector& detector = m_detectorsPackPimpl->suspensionDetector;
                return handleEvent(detector, session, &packet, tlsData);
            }

            Violation Manager::onTCPReset(Session &session, const TLSData &tlsData) noexcept
            {
                Detector& detector = m_detectorsPackPimpl->handshakeAgingTimeDetector;
                return handleEvent(detector, session, nullptr, tlsData);
            }

            Violation Manager::onSessionEnd(Session &session, const TLSData &tlsData) noexcept
            {
                Detector& detector = m_detectorsPackPimpl->finishDetector;
                
                return handleEvent(detector, session, nullptr, tlsData);
            }

            {
                return m_sessionParamsViewer->getPacketAction();
            }

TLSFlow::PacketFlow::Engine::Action PacketFlowManager::onParsedClientHello(Session &session, DPFrame &packet, const TLSFlow::Fingerprint &fingerprint)
{
    /* Fetch policy */
    DPPolicyItem *policy = DPSharedDoublePolicyArray::getGlobalSharedPolicyArray()->getActivePolicy(packet.getPolicyId());
    if(policy == nullptr || !policy->profileExists[IDS_PROFILE_TLS_FINGERPRINT])
        return TLSFlow::PacketFlow::Engine::Action::Continue;

    /* Fetch profile */
    SharedProfile profile;
    if(m_pConfigurationManager->getProfile(packet.getPolicyId(), profile) != NSConfigurationManager::Status::Ok)
        return TLSFlow::PacketFlow::Engine::Action::Continue;

    /* Build server key */
    NSServerManager::Engine::ServerKey serverKey(policy->hashKey, packet.getPayloadDstIpAddr(), packet.getPayloadDstPort());

    if (m_pServerManager->shouldBeInspected(serverKey) == false)
        return TLSFlow::PacketFlow::Engine::Action::Continue;

    /* Build key */
    MasterKey key(policy->hashKey, profile.getHash(), fingerprint);
    NSStatisticsManager::Common::PolicyServer server(policy->hashKey, packet.getPayloadDstIpAddr(), packet.getPayloadDstPort());

    /* Find filter */
    ClassificationResult result;
    ClassificationStatus status = m_pFilters->classify(key, result);

    /* Should we save full string */
    bool saveFullstring = true;

    /* Update statistics */
    uint64_t count = 0;
    Action effectiveAction;
    if(status == ClassificationStatus::Classified)
    {
        if(result.getType() == FilterType::Manual)
        {
            saveFullstring = false;
            m_pStatistics->increment({ packet.getPolicyId(), key, NSStatisticsManager::Common::PolicyServer(), 1 /* Hit */, 1 /* Manual Filter Hit */, 0 /* Behavioral Filter Hit */, 0 /* Reputation Filter Hit */, 0 /* Randomized Filter Hit */, 0 /* On-Demand Filter Hit */});
            count = m_pStatistics->increment( { NSStatisticsManager::Engine::AttackType::Manual, packet.getPolicyId(), key, 1, packet.getFrameLenWithoutInternalTags() }).second;
        }
        else if(result.getType() == FilterType::Behavioral)
        {
            m_pStatistics->increment({ packet.getPolicyId(), key, server, 1 /* Hit */, 0 /* Manual Filter Hit */, 1 /* Behavioral Filter Hit */, 0 /* Reputation Filter Hit */, 0 /* Randomized Filter Hit */, 0 /* On-Demand Filter Hit */ });
            count = m_pStatistics->increment({ NSStatisticsManager::Engine::AttackType::Behavioral, packet.getPolicyId(), key, 1, packet.getFrameLenWithoutInternalTags() }).second;
        }
        else if(result.getType() == FilterType::Reputation)
        {
            saveFullstring = false;
            m_pStatistics->increment({ packet.getPolicyId(), key, NSStatisticsManager::Common::PolicyServer(), 1 /* Hit */, 0 /* Manual Filter Hit */, 0 /* Behavioral Filter Hit */, 1 /* Reputation Filter Hit */, 0 /* Randomized Filter Hit */, 0 /* On-Demand Filter Hit */});
            count = m_pStatistics->increment({ NSStatisticsManager::Engine::AttackType::Reputation, packet.getPolicyId(), key, 1, packet.getFrameLenWithoutInternalTags() }).second;
        }
        else if(result.getType() == FilterType::Randomized)
        {
            if(result.getAction() == Action::Allow) {
                m_pStatistics->increment({ packet.getPolicyId(), key, server, 1 /* Hit */, 0 /* Manual Filter Hit */, 0 /* Behavioral Filter Hit */, 0 /* Reputation Filter Hit */, 1 /* Randomized Filter Hit */, 0 /* On-Demand Filter Hit */});
            }
            else {
                m_pStatistics->increment({ packet.getPolicyId(), key, server, 1 /* Hit */, 0 /* Manual Filter Hit */, 0 /* Behavioral Filter Hit */, 0 /* Reputation Filter Hit */, 0 /* Randomized Filter Hit */, 0 /* On-Demand Filter Hit */});
                count = m_pStatistics->increment({ NSStatisticsManager::Engine::AttackType::Randomized, packet.getPolicyId(), key, 1, packet.getFrameLenWithoutInternalTags() }).second;
            }
        }
        else if(result.getType() == FilterType::OnDemand)
        {
            m_pStatistics->increment({ packet.getPolicyId(), key, server, 1 /* Hit */, 0 /* Manual Filter Hit */, 0 /* Behavioral Filter Hit */, 0 /* Reputation Filter Hit */, 0 /* Randomized Filter Hit */, 1 /* On-Demand Filter Hit */});
            count = m_pStatistics->increment({ NSStatisticsManager::Engine::AttackType::OnDemand, packet.getPolicyId(), key, 1, packet.getFrameLenWithoutInternalTags() }).second;
        }

        /* Set effective action for filter */
        if(result.getAction() == Action::AsProfile)
            effectiveAction = profile.getAction();
        else
            effectiveAction = result.getAction();
    }
    else
        m_pStatistics->increment({ packet.getPolicyId(), key, server, 1 /* Hit */, 0 /* Manual Filter Hit */, 0 /* Behavioral Filter Hit */, 0 /* Reputation Filter Hit */, 0 /* Randomized Filter Hit */, 0 /* On-Demand Filter Hit */});

    if(saveFullstring)
    {
        /* add to full string table */
        if(! m_pFullStringProcessor->isExist(fingerprint))
        {
            const auto& fs = TLSFlow::Context::Engine::Application::getInstance().getClientHelloDataViewer()->getFullString();
            if(fs.getLength() > 0)
            {
                if(m_pFullStringProcessor->insert(fingerprint, fs.getData(), fs.getLength()) != IFullStringProcessor::ProcessResult::Success)
                {
                    ostringstream fpstr;
                    fpstr << fingerprint;
                    char *fpch = (char *)fpstr.str().c_str();
                    //log error here
                    DIAG_LOGGER_INJECT(TLSFP_FINGERPRINT_FULLSTRING_INSERT_ERROR, fpch);
                }
            }
        }
    }

    /* Report */
    if(status == ClassificationStatus::Classified && effectiveAction != Action::Allow) {
        AttackType attackType;
        if(result.getType() == FilterType::Manual)
            attackType = AttackType::Manual;
        else if(result.getType() == FilterType::Behavioral)
            attackType = AttackType::Behavioral;
        else if(result.getType() == FilterType::Reputation)
            attackType = AttackType::Reputation;
        else if(result.getType() == FilterType::OnDemand)
            attackType = AttackType::OnDemand;
        else
            attackType = AttackType::Randomized;

        m_pReportingHandler->handleReport(attackType, &packet, key, count);
    }

    /* Mitigate */
    if  (status == ClassificationStatus::Classified && (policy->action == IDS_policyAction_blockNreport_E && effectiveAction == Action::BlockAndReport))
    {
        RTPCprotection rtpcModule;
        if(result.getType() == FilterType::Manual)
            rtpcModule = RTPC_TLS_FINGERPRINT_FILTERS;
        else if(result.getType() == FilterType::Behavioral)
            rtpcModule = RTPC_TLS_FINGERPRINT_BEHAVIORAL;
        else if(result.getType() == FilterType::Reputation)
            rtpcModule = RTPC_TLS_FINGERPRINT_REPUTATION;
        else if(result.getType() == FilterType::OnDemand)
            rtpcModule = RTPC_TLS_FINGERPRINT_ONDEMAND;
        else
            rtpcModule = RTPC_TLS_FINGERPRINT_RANDOMIZED;

        if (session.isAxSession())
        {
            packet.packetActionSet(PACKET_ACTION_REVERSE_RST_BACK, rtpcModule);
            session.setIsFeResetSent(true);
            session.setAgingTimeout(1);
        }
        else
        {
            // Send rst both to client and to server only is session was already established with server
            if (session.getState() == session_state_established)
                packet.packetActionSet(PACKET_ACTION_RESET_BIDIRECTIONAL, rtpcModule);
            else
                packet.packetActionSet(PACKET_ACTION_REVERSE_RST_BACK, rtpcModule);

            session.setAgingTimeout(1);

        }

        return TLSFlow::PacketFlow::Engine::Action::Stop;
    }

    return TLSFlow::PacketFlow::Engine::Action::Continue;
}

void TLSMonitor::collect(const Params& params) noexcept
{
    /*
     * Step 1: Determine SNI type from configuration (full or base). If base, extract base SNI from params.
     * Step 2: Log the parameters.
     *  Step 2.1: Log Server (policy ID, IP, port) -> get server ID.
     *  Step 2.2: Log SNI (policy ID, Base-SNI/Full-SNI) -> get SNI ID
     * Step 3: Record the parameters.
     */

    // Step 1:
    // Determine SNI type from configuration
    SniType sniType;
    auto policyId = params.getPolicyId();
    if(policyId == NSCommon::InvalidPolicyId)
    {
        // Invalid policy ID, abort
        return;
    }
    m_pConfiguration->getSettings(policyId, [&sniType](NSCommon::PolicyId, const NSConfiguration::Settings& settings) {
        sniType = settings.getSniType();
    });


    //Step 2: Log the parameters.
    //Log Server:
    auto serverId = m_pServerRegistry->log({policyId, params.getIP(), params.getPort()});
    if(serverId == NSCommon::InvalidServerId)
    {
        // Failed to log server, abort
        return;
    }
    // Log SNI:
    // If SNI type is Base, extract base SNI from params
    NSCommon::SniId sniId{NSCommon::InvalidSniId};
    if(sniType == SniType::Base)
    {
        sniId = m_pServerRegistry->log({policyId, params.getSni().getLen() == 0 ? TLSFlow::Sni{"NoSniExtension"} : TLSFlow::Sni{FQDN::Manager::getInstance().getBaseDomain(params.getSni().c_str())}});
    }
    else
    {
        sniId = m_pServerRegistry->log({policyId, params.getSni().getLen() == 0 ? TLSFlow::Sni{"NoSniExtension"} : params.getSni()});
    }

    if(sniId == NSCommon::InvalidSniId)
    {
        // Failed to log SNI, abort
        return;
    }

    //Step 3: Record the parameters. build params with constructor Params(NSCommon::PolicyId policyId, NSCommon::ServerId serverId, NSCommon::SniId sniId, const TLSFlow::Fingerprint& fingerprint, Misc::TLSVersion version)
    m_pRecorder->record({policyId, serverId, sniId, params.getFingerprint(), params.getTlsVersion()});
}
