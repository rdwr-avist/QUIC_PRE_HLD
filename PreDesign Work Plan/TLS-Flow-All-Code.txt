#pragma once

#include <cstdint>
#include <functional>
#include <misc/U128.hpp>
#include <iomanip>
#include "Definitions.hpp"

using namespace std;
using namespace Misc;

namespace TLSFlow
{
    class Fingerprint
    {
    private:
        /**
         * Members
         */
        U128 m_value;
    public:
        /**
         * Param constructor
         *
         * @param hash
         */
        Fingerprint(): m_value(0, 0) {};
        Fingerprint(const U128& value) : m_value(value) { };

        Fingerprint(const char *fp) {
            m_value = 0;
            string fpStr = fp;
            if (FingerprintByteLength == 16 && fpStr.length() == FingerprintByteLength * 2) {
                std::string high_part = fpStr.substr(0, 16);
                std::string low_part = fpStr.substr(16);
                try {
                    m_value.high = std::stoull(high_part, nullptr, 16);
                }
                catch (const std::exception& e) {m_value = 0;}
                try {
                    m_value.low = std::stoull(low_part, nullptr, 16);
                }
                catch (const std::exception& e) {m_value = 0;}
            }
        }

        /**
         * Getters
         */
        const U128& value() const { return m_value; }
        U128& value() { return m_value; }

        void setValue(const U128 &mValue) {
            m_value = mValue;
        }

        __uint128_t getFingerprintUint128() const {
            __uint128_t val = m_value.high;
            val <<= 64;
            val += m_value.low;
            return val;
        }
        void setFingerprintUint128(const __uint128_t &mFingerprint) {
            m_value.high = static_cast<uint64_t>(mFingerprint >> 64);
            m_value.low = static_cast<uint64_t>(mFingerprint);
        }

        /**
         * Operators
         */
        bool operator==(const Fingerprint& other) const { return m_value == other.m_value; }
        bool operator==(const U128& value) const { return m_value == value; }
        bool operator!=(const Fingerprint& other) const { return m_value != other.m_value; }
        bool operator!=(const U128& value) const { return m_value != value; }
//        void operator=(const Fingerprint& other) { m_value = other.m_value; }
        Fingerprint& operator=(const Fingerprint& other) { m_value = other.m_value; return *this;}
        void operator=(const U128& value) { m_value = value; }
        friend ostream& operator<<(ostream& output, const Fingerprint& fingerprint) { output << std::setw(16) << std::setfill('0') << std::hex << std::uppercase << fingerprint.value().high; output << std::setw(16) << std::setfill('0') << std::hex << std::uppercase << fingerprint.value().low; return output; }
        operator string() const { stringstream output; output << *this; return output.str(); }
    };
};

/*******************************************
 * Hash injection into std namespace
 *******************************************/
namespace std
{
    template<> struct hash<TLSFlow::Fingerprint>
    {
        auto operator()(const TLSFlow::Fingerprint& fingerprint) const
        {
            return hash<size_t>()(hash<uint64_t>()(fingerprint.value().high) + hash<uint64_t>()(fingerprint.value().low));
        }
    };
}

namespace TLSFlow {
    static constexpr uint16_t max_sni_length = 255;

     class Sni {

    public:
        Sni(): m_len{0} {}
        void clear() { sni[0] = 0; m_len = 0;}

        // Constructor from a C-style string
        Sni(const char* sniCStr):m_len{0} {
            if (sniCStr) {
                m_len = strlen(sniCStr);
                if (m_len > max_sni_length) {
                    m_len = max_sni_length;
                }
                memcpy(this->sni, sniCStr, m_len);
            }
            this->sni[m_len] = 0; // Null terminate
        }

        //Copy constructor
        Sni(const Sni &other) {
            m_len = other.m_len;
            memcpy(sni, other.sni, m_len);
            sni[m_len] = 0; // Null terminate
        }

        //Copy assignment operator
        Sni& operator=(const Sni &other) {
            if (this != &other) {
                m_len = other.m_len;
                memcpy(sni, other.sni, m_len);
                sni[m_len] = 0; // Null terminate
            }
            return *this;
        }

        const char* c_str() const { return sni; }

        bool set(const char *ptr, uint16_t len) {

            if (len > max_sni_length){
                return false;
            }
            m_len = len;

            memcpy(sni, ptr, len);
            sni[len] = 0;
            return true;
        }

        bool set(const string &in) {
            return set(in.c_str(), in.length());
        }

        bool operator==(const Sni &other) const { return (m_len == other.m_len) && (memcmp(sni, other.sni, m_len) == 0); }
        void getCopy(char* out) const{
            if (out) {
                memcpy(out, sni, m_len + 1);
         }
        }
        bool isEmpty() const { return (m_len == 0);}
        int  getLen() const { return m_len;}
        friend ostream& operator<<(ostream& output, const Sni& sni) { output << sni.sni; return output; }
        operator string() const { return string(sni, m_len); }
    private:
        char sni[max_sni_length + 1];
        size_t  m_len; // Not including null termination
    };
}

namespace std
{
    template<> struct hash<TLSFlow::Sni>
    {
        size_t operator()(const TLSFlow::Sni& key) const
        {
            char sniStr[TLSFlow::max_sni_length + 1];
            key.getCopy(sniStr);

            // round up to the next 4-byte boundary and pad with zeros
            uint32_t length =  key.getLen();
            length = (length + 3) & ~3;
            memset(sniStr + key.getLen() + 1, 0, length - key.getLen());

            return idsp_hash_func(sniStr, length, MAX_UINT_32);
        }
    };
}

#pragma once

#include <cstdint>
#include "TLSData.hpp"

namespace TLSFlow
{
    constexpr uint8_t TLSAlertWarningLevel = 1;
    constexpr uint8_t TLSAlertFatalLevel = 2;
    constexpr uint8_t TLSAlertCloseNotify = 0;
    constexpr uint8_t TLSAlertUserCanceled = 90;

    inline bool isTerminatingAlert(uint8_t tlsVersionMask,
                                   uint8_t alertLevel,
                                   uint8_t alertDescription)
    {
        if (alertLevel != TLSAlertWarningLevel && alertLevel != TLSAlertFatalLevel) {
            return false;
        }

        if (alertDescription == TLSAlertCloseNotify) {
            return true;
        }

        const uint8_t knownVersionMask = static_cast<uint8_t>(TLS12OrLessBitMask | TLS13BitMask);
        const bool tls12OrLessOnly = isTLS12OrLess(tlsVersionMask) &&
                                     !isTLS13(tlsVersionMask) &&
                                     (tlsVersionMask & static_cast<uint8_t>(~knownVersionMask)) == 0;
        const bool tls13Only = tlsVersionMask == TLS13BitMask;

        if (tls13Only) {
            return alertDescription != TLSAlertUserCanceled;
        }

        if (tls12OrLessOnly) {
            return alertLevel == TLSAlertFatalLevel;
        }

        return alertLevel == TLSAlertFatalLevel &&
               alertDescription != TLSAlertUserCanceled;
    }
}

#pragma once

#include <cstdint>
#include "TLSViolation.hpp"
#include "../../common/include/Fingerprint.hpp"

namespace TLSFlow
{
  /**
    * @brief TLS handshake lifecycle state.
    */
    enum class HsState : uint8_t
    {
        PreHandshake,
        InProgress,
        ChangeCipherSpec,
        Finished,
		Error,
		TerminatedByAlert,
		Unsupported,
    };

	static_assert(static_cast<uint8_t>(HsState::ChangeCipherSpec) <
	              static_cast<uint8_t>(HsState::Finished),
	              "HsState ordering is required by the encrypted handshake boundary");
	static_assert(static_cast<uint8_t>(HsState::Unsupported) >
			      static_cast<uint8_t>(HsState::Finished),
	              "HsState ordering is required by terminal-state checks");
	static_assert(static_cast<uint8_t>(HsState::TerminatedByAlert) >
	              static_cast<uint8_t>(HsState::Finished),
	              "HsState ordering is required by terminal-state checks");

	inline bool isActiveHandshakeState(HsState state)
	{
		return state == HsState::PreHandshake ||
		       state == HsState::InProgress ||
		       state == HsState::ChangeCipherSpec;
	}

	inline bool isTerminalState(HsState state)
	{
		return state == HsState::Finished ||
		       state == HsState::Error ||
		       state == HsState::Unsupported ||
		       state == HsState::TerminatedByAlert;
	}

	inline bool isSilentTerminalState(HsState state)
	{
		return state == HsState::Finished || state == HsState::Unsupported;
	}

	/**
	* @brief Bit positions for the tlsVersionMask field in TLSData.
	*
	* Each enumerator is the shift amount for the corresponding TLS version bit.
	* Example: (1u << static_cast<uint8_t>(TLSVersionBit::TLS13)) selects the TLS 1.3 bit.
	*/
	enum class TLSVersionBit : uint8_t
	{
		SSL30 = 0,
		TLS10 = 1,
		TLS11 = 2,
		TLS12 = 3,
		TLS13 = 4
	};

	constexpr uint8_t TLS13BitMask = 1u << static_cast<uint8_t>(TLSVersionBit::TLS13);
	constexpr uint8_t TLS12OrLessBitMask = TLS13BitMask - 1u;

	/**
	* @brief Returns true if the TLS 1.3 bit is set in the given version mask.
	*
	* @param versionMask  Bitmask where each bit position is defined by TLSVersionBit.
	*/
	inline bool isTLS13(uint8_t versionMask)
	{
		return versionMask & TLS13BitMask;
	}

	/**
	* @brief Returns true if the TLS 1.2 or lower bit is set in the given version mask.
	*
	* @param versionMask  Bitmask where each bit position is defined by TLSVersionBit.
	*/
	inline bool isTLS12OrLess(uint8_t versionMask)
	{
		return versionMask & TLS12OrLessBitMask;
	}
	
	/**
	* @brief Per-session TLS flow context, owned by the TLS state update layer.
	*
	* Holds all mutable per-session state accumulated during TLS parsing.
	* All fields are reset to their initial defaults by clear().
	*
	* ch = ClientHello
	* ccs = ChangeCipherSpec
	* AppData = ApplicationData
	*/
	struct TLSData
	{
		HsState          hsState;
		uint8_t          tlsVersionMask;
		uint8_t          clientHelloCount;
		uint8_t          earlyDataCount;
		bool             earlyDataOffered;
		uint8_t          preCcsSeen;
		TLSFlow::Fingerprint chFingerprint;
		TLSViolationMask terminalViolationMask;
		unsigned long clientHelloTimeStamp;
	
		TLSData()
			: hsState(HsState::PreHandshake)
			, tlsVersionMask(0)
			, clientHelloCount(0)
			, earlyDataCount(0)
			, earlyDataOffered(false)
			, preCcsSeen(0)
			, chFingerprint()
			, terminalViolationMask(TLSViolationMaskNone),
			clientHelloTimeStamp(0)
		{}
	
		/**
		* @brief Sets the bit for the given TLS version in tlsVersionMask.
		*
		* @param version  Bit position to set, as defined by TLSVersionBit.
		*/
		void enableVersion(TLSVersionBit version)
		{
			tlsVersionMask |= static_cast<uint8_t>(1u << static_cast<uint8_t>(version));
		}
	
		/**
		* @brief Clears the bit for the given TLS version in tlsVersionMask.
		*
		* @param version  Bit position to clear, as defined by TLSVersionBit.
		*/
		void disableVersion(TLSVersionBit version)
		{
			tlsVersionMask &= static_cast<uint8_t>(~(1u << static_cast<uint8_t>(version)));
		}
	
		/**
		* @brief Resets all fields to their initial defaults.
		*
		* Idempotent — calling clear() on an already-default TLSData is safe.
		*/
		void clear()
		{
			hsState               = HsState::PreHandshake;
			tlsVersionMask        = 0;
			clientHelloCount      = 0;
			earlyDataCount        = 0;
			earlyDataOffered     = false;
			preCcsSeen            = 0;
			chFingerprint    = TLSFlow::Fingerprint();
			terminalViolationMask = TLSViolationMaskNone;
			clientHelloTimeStamp   = 0;
		}
	};

	inline bool usesTerminalPayloadPath(HsState state)
	{
		return state == HsState::Error ||
		       state == HsState::TerminatedByAlert;
	}
	
}

#pragma once

#include <cstdint>

namespace TLSFlow
{
    enum class TLSViolation : uint8_t
    {
        None = 0,  // no violation; verdict is non-terminal

        // ── TLS-Enforcement violations ─────────────────────────────────────────
        TLSEnforcementMalformedTLSHandshake,    // generic handshake failure
        TLSEnforcementWrongVersion,             // negotiated TLS version does not meet policy requirement
        TLSEnforcementMaxCiphers,               // cipher suite count exceeds configured maximum
        TLSEnforcementMaxCompression,           // compression method count exceeds configured maximum
        TLSEnforcementMaxExtensions,            // extension count exceeds configured maximum
        TLSEnforcementPrematureClosure,         // TCP-Reset immediately after CH
        TLSEnforcementIncompleteHandshake,      // session closed before handshake completed
        TLSEnforcementEarlyDataExceeded,        // AppData before CH exceeds allowed count (TLS 1.3)
        TLSEnforcementUnexpectedPreCCS,         // excess CCS records before CH
        TLSEnforcementAlertDuringHandshake,     // clear Alert record seen during HS
        TLSEnforcementMalformedClientHello,     // ClientHello structure could not be parsed
        TLSEnforcementUnexpectedRecord,         // record type not permitted in current HS state
        TLSEnforcementMultipleClientHello,      // second ClientHello not permitted (TLS ≤1.2)

        TLSMaxViolationTypes // Must be last. not a real violation
        
    };

    using TLSViolationMask = uint32_t;
    constexpr TLSViolationMask TLSViolationMaskNone = 0u;

    static_assert(static_cast<uint8_t>(TLSViolation::TLSMaxViolationTypes) < 32u,
                  "TLSViolationMask width must fit all TLSViolation values");

    inline bool isValidTLSViolation(TLSViolation violation) noexcept
    {
        return violation != TLSViolation::None &&
               violation != TLSViolation::TLSMaxViolationTypes;
    }

    inline TLSViolationMask toTLSViolationMask(TLSViolation violation) noexcept
    {
        if (!isValidTLSViolation(violation)) {
            return TLSViolationMaskNone;
        }

        return static_cast<TLSViolationMask>(1u << static_cast<uint8_t>(violation));
    }

    inline bool hasTLSViolation(TLSViolationMask mask, TLSViolation violation) noexcept
    {
        return (mask & toTLSViolationMask(violation)) != 0u;
    }

    inline TLSViolation highestTLSViolation(TLSViolationMask mask) noexcept
    {
        for (int violation = static_cast<int>(TLSViolation::TLSMaxViolationTypes) - 1;
             violation > static_cast<int>(TLSViolation::None);
             --violation) {
            const TLSViolation current = static_cast<TLSViolation>(violation);
            if (hasTLSViolation(mask, current)) {
                return current;
            }
        }

        return TLSViolation::None;
    }
} // TLSFlow

#include "../include/Application.hpp"
#include "../../common/include/InitException.hpp"
#include <diag/diag_log_inject.h>

using namespace TLSFlow;
using namespace TLSFlow::Context;
using namespace TLSFlow::Context::Controller;

Application* Application::instance = nullptr;

bool Application::init(InitPhase phase, std::unique_ptr<IFactory> factory)
{
    switch(phase)
    {
        case Controller_E:
        {
            /* Validate */
            if(factory == nullptr)
            {
                DIAG_LOGGER_INJECT(TLS_FLOW_INIT);
                return false;
            }

            /* Save factory */
            m_pFactory = std::move(factory);

            /* Init */
            try
            {
                m_pFactory->initController();
            }
            catch (const InitException& e)
            {
                DIAG_LOGGER_INJECT(TLS_FLOW_COMPONENTS_INIT, e.getPhase().c_str(), e.getModule().c_str(), e.getMessage().c_str());
                return false;
            }
            break;
        }
        case Legacy_E:
        {
            if(m_pFactory == nullptr)
            {
                DIAG_LOGGER_INJECT(TLS_FLOW_INIT);
                return false;
            }

            try
            {
                m_pFactory->initLegacy();
            }
            catch (const InitException& e)
            {
                DIAG_LOGGER_INJECT(TLS_FLOW_COMPONENTS_INIT, e.getPhase().c_str(), e.getModule().c_str(), e.getMessage().c_str());
                return false;
            }
            break;
        }
        default:
            break;
    }
    return true;
}

Application &Application::getInstance() {
    if (instance == nullptr) {
        instance = new Application();
    }
    return *instance;
}

void Application::destroyInstance() {
    if (instance != nullptr) {
        delete instance;
        instance = nullptr;
    }
}

std::shared_ptr<NSDiagnostics::Controller::IDiagnosticsCounters> Application::getDiagnosticsApi()
{
    return m_pFactory->getDiagnosticsCounters();
}

std::shared_ptr<NSDiagnostics::Controller::TLSSessionExtensionInspector>
Application::getSessionExtensionInspector()
{

void Factory::buildTLSFlowConductor()
{
    /* Init session observer */
    auto extensionTable = std::make_unique<TLSFlow::TlsSessionExtensionTable>();
    tlsFlowConductor = std::make_shared<PacketFlow::Engine::TLSFlowConductor>(std::move(extensionTable));
    if (tlsFlowConductor == nullptr)
        throw InitException("Engine Init", "Packet-Flow", "Failed to create session observer object");

    /* Register to session global observer */
    if (sessionTable->registerGlobalObserver(tlsFlowConductor.get(), STM_EVENT_NEW_SESSION, NewSessionPriorityTLS) != RET_STM_SUCCESS)
        throw InitException("Engine Init", "Packet-Flow", "Failed to register connection observer to session table");
}

    auto diagnosticsCountersTable = make_unique<NSDiagnostics::Engine::DiagnosticsCountersTable>();
    if(diagnosticsCountersTable->init(sharedMemoryManager, coreId, MemoryChunksID::MEMORY_INSTANCE_SHARED_TLS_FLOW_DIAGNOSTICS_FIRST_ID, MemoryChunksID::MEMORY_INSTANCE_SHARED_TLS_FLOW_DIAGNOSTICS_LAST_ID - 1, to_underlying(NSDiagnostics::Common::DiagnosticsEvents::MaxEventId)) != NSCounterTable::Status::Ok)

constexpr uint32_t ResponseTimeoutMs = 10000;

bool compareEngineIds(const std::string& first, const std::string& second)
{
    return std::atoi(first.c_str()) < std::atoi(second.c_str());
}

std::string hsStateToString(uint8_t state)
{
    switch (static_cast<TLSFlow::HsState>(state))
    {
        case TLSFlow::HsState::PreHandshake: return "PreHandshake";
        case TLSFlow::HsState::InProgress: return "InProgress";
        case TLSFlow::HsState::ChangeCipherSpec: return "ChangeCipherSpec";
        case TLSFlow::HsState::Finished: return "Finished";
        case TLSFlow::HsState::Error: return "Error";
        case TLSFlow::HsState::TerminatedByAlert: return "TerminatedByAlert";
        case TLSFlow::HsState::Unsupported: return "Unsupported";
        default: return "Unknown";
    }
}

std::string tlsVersionMaskToString(uint8_t mask)
{
    static const char* names[] = {"SSL3.0", "TLS1.0", "TLS1.1", "TLS1.2", "TLS1.3"};
    std::ostringstream output;
    bool first = true;
    for (uint8_t bit = 0; bit < 5; ++bit)
    {
        if ((mask & static_cast<uint8_t>(1u << bit)) != 0)
        {
            if (!first)
            {
                output << ',';
            }
            output << names[bit];
            first = false;
        }
    }
    return first ? "None" : output.str();
}

std::string violationToString(uint8_t violation)
{
    switch (static_cast<TLSFlow::TLSViolation>(violation))
    {
        case TLSFlow::TLSViolation::None: return "None";
        case TLSFlow::TLSViolation::TLSEnforcementMalformedTLSHandshake: return "MalformedHandshake";
        case TLSFlow::TLSViolation::TLSEnforcementWrongVersion: return "WrongVersion";
        case TLSFlow::TLSViolation::TLSEnforcementMaxCiphers: return "MaxCiphers";
        case TLSFlow::TLSViolation::TLSEnforcementMaxCompression: return "MaxCompression";
        case TLSFlow::TLSViolation::TLSEnforcementMaxExtensions: return "MaxExtensions";
        case TLSFlow::TLSViolation::TLSEnforcementPrematureClosure: return "PrematureClosure";
        case TLSFlow::TLSViolation::TLSEnforcementIncompleteHandshake: return "IncompleteHandshake";
        default: return "Unknown";
    }
}

std::string fingerprintToString(const IpcMsgTlsSessionExtensionTableRecord& record)
{
    std::ostringstream output;
    output << std::uppercase << std::hex << std::setfill('0')
           << std::setw(16) << record.m_fingerprintHigh
           << std::setw(16) << record.m_fingerprintLow;
    return output.str();
}

TLSSessionExtensionInspector::TLSSessionExtensionInspector(
    IpcMessageClass* ipcMessage,
    uint32_t numOfCores)
    : m_ipcMessage(ipcMessage), m_numOfCores(numOfCores), m_pageState(numOfCores + 1)
{
}

STATUS TLSSessionExtensionInspector::init()
{
    if (m_ipcMessage == nullptr)
    {
        return ERROR;
    }

    char semaphoreName[] = "tls_session_extension_inspector";
    OSITCG_sem_create(semaphoreName, 1, &m_responseSem);

    const auto sizeStatus = m_ipcMessage->registerObserver(
        IPC_MSG_TYPE_TLS_SESSION_EXTENSION_TABLE_SIZE_RESPONSE, this);
    const auto fetchStatus = m_ipcMessage->registerObserver(
        IPC_MSG_TYPE_TLS_SESSION_EXTENSION_TABLE_FETCH_RESPONSE, this);
    if (sizeStatus != OK || fetchStatus != OK)
    {
        return ERROR;
    }

    const std::vector<uint8_t> sizeWidths{8, 12};
    const std::vector<std::string> sizeNames{"Engine", "Size"};
    m_sizeTablePrinter.init(&sizeWidths, &sizeNames);

    const std::vector<uint8_t> entryWidths{8, 12, 18, 18, 20};
    const std::vector<std::string> entryNames{"Engine", "Session", "State", "TLS Versions", "Violation"};
    m_entryTablePrinter.init(&entryWidths, &entryNames);
    return OK;
}

void TLSSessionExtensionInspector::handleMsg(IpcMsgData& ipcMsg)
{
    switch (ipcMsg.m_msgType)
    {
        case IPC_MSG_TYPE_TLS_SESSION_EXTENSION_TABLE_SIZE_RESPONSE:
            handleSizeResponse(ipcMsg);
            break;
        case IPC_MSG_TYPE_TLS_SESSION_EXTENSION_TABLE_FETCH_RESPONSE:
            handleFetchResponse(ipcMsg);
            break;
        default:
            break;
    }
}

bool TLSSessionExtensionInspector::requestSize(uint32_t coreId)
{
    if (coreId > m_numOfCores)
    {
        return false;
    }

    resetRequestState();
    m_requestKind = RequestKind::Size;
    m_requestedCore = coreId;
    m_pendingResponses = coreId == 0 ? m_numOfCores : 1;
    m_requestActive = true;
    ++m_requestId;
    OSITCG_sem_p(m_responseSem);

    if (!sendSizeRequest(coreId))
    {
        m_requestActive = false;
        m_pendingResponses = 0;
        OSITCG_sem_v(m_responseSem);
        return false;
    }
    return waitForResponses();
}

bool TLSSessionExtensionInspector::requestEntries(uint32_t coreId,
                                                   bool allSessions,
                                                   int32_t sessionIndex,
                                                   uint32_t maxEntries)
{
    if (coreId > m_numOfCores || maxEntries == 0)
    {
        return false;
    }

    resetRequestState();
    m_requestKind = RequestKind::Entries;
    m_requestedCore = coreId;
    m_allSessions = allSessions;
    m_sessionIndex = sessionIndex;
    m_pendingResponses = coreId == 0 ? m_numOfCores : 1;
    m_requestActive = true;
    ++m_requestId;
    for (uint32_t engine = 1; engine <= m_numOfCores; ++engine)
    {
        m_pageState[engine].remaining = maxEntries;
    }
    OSITCG_sem_p(m_responseSem);

    if (!sendFetchRequest(coreId, false, 0, maxEntries))
    {
        m_requestActive = false;
        m_pendingResponses = 0;
        OSITCG_sem_v(m_responseSem);
        return false;
    }
    return waitForResponses();
}

bool TLSSessionExtensionInspector::sendSizeRequest(uint32_t coreId)
{
    IpcMsgData message{};
    message.m_destMask = IPC_SHARED_MEMORY_BROADCAST_INDEX;
    message.m_msgType = IPC_MSG_TYPE_TLS_SESSION_EXTENSION_TABLE_SIZE_REQUEST;
    message.m_data.m_tlsSessionExtensionTableSizeRequest.m_coreId = coreId;
    message.m_data.m_tlsSessionExtensionTableSizeRequest.m_requestId = m_requestId;
    return m_ipcMessage->sendMsg(message) == QUEUE_OK;
}

bool TLSSessionExtensionInspector::sendFetchRequest(uint32_t coreId,
                                                     bool hasCursor,
                                                     uint32_t cursor,
                                                     uint32_t maxEntries)
{
    IpcMsgData message{};
    message.m_destMask = IPC_SHARED_MEMORY_BROADCAST_INDEX;
    message.m_msgType = IPC_MSG_TYPE_TLS_SESSION_EXTENSION_TABLE_FETCH_REQUEST;
    message.m_data.m_tlsSessionExtensionTableFetchRequest.m_coreId = coreId;
    message.m_data.m_tlsSessionExtensionTableFetchRequest.m_allSessions = m_allSessions;
    message.m_data.m_tlsSessionExtensionTableFetchRequest.m_sessionIndex = m_sessionIndex;
    message.m_data.m_tlsSessionExtensionTableFetchRequest.m_maxEntries = maxEntries;
    message.m_data.m_tlsSessionExtensionTableFetchRequest.m_hasCursor = hasCursor;
    message.m_data.m_tlsSessionExtensionTableFetchRequest.m_cursor = cursor;
    message.m_data.m_tlsSessionExtensionTableFetchRequest.m_requestId = m_requestId;
    return m_ipcMessage->sendMsg(message) == QUEUE_OK;
}

bool TLSSessionExtensionInspector::waitForResponses()
{
    const bool completed = OSITCG_sem_p_wait(
        m_responseSem, OSTIMG_ms_to_tick(ResponseTimeoutMs));
    const bool success = completed && m_pendingResponses == 0;
    if (!completed)
    {
        m_pendingResponses = 0;
    }
    m_requestActive = false;
    OSITCG_sem_v(m_responseSem);
    return success;
}

bool TLSSessionExtensionInspector::isSelectedCore(uint32_t coreId) const
{
    return coreId > 0 && coreId <= m_numOfCores &&
           (m_requestedCore == 0 || m_requestedCore == coreId);
}

void TLSSessionExtensionInspector::completeEngine(uint32_t coreId)
{
    if (coreId > m_numOfCores || m_pageState[coreId].complete)
    {
        return;
    }
    m_pageState[coreId].complete = true;
    if (m_pendingResponses > 0)
    {
        --m_pendingResponses;
    }
    signalIfComplete();
}

void TLSSessionExtensionInspector::signalIfComplete()
{
    if (m_pendingResponses == 0)
    {
        OSITCG_sem_v(m_responseSem);
    }
}

void TLSSessionExtensionInspector::resetRequestState()
{
    m_requestActive = false;
    m_requestKind = RequestKind::None;
    m_requestedCore = 0;
    m_pendingResponses = 0;
    for (auto& state : m_pageState)
    {
        state = EnginePageState{};
    }
    m_sizeTablePrinter.clearTableRows();
    m_entryTablePrinter.clearTableRows();
    m_entryRows.clear();
}

void TLSSessionExtensionInspector::handleSizeResponse(const IpcMsgData& ipcMsg)
{
    if (!m_requestActive || m_requestKind != RequestKind::Size)
    {
        return;
    }

    const auto& response = ipcMsg.m_data.m_tlsSessionExtensionTableSizeResponse;
    if (response.m_requestId != m_requestId || !isSelectedCore(response.m_coreId))
    {
        return;
    }

    if (m_pageState[response.m_coreId].complete)
    {
        return;
    }

    m_sizeTablePrinter.addTableRow({std::to_string(response.m_coreId), std::to_string(response.m_size)});
    completeEngine(response.m_coreId);
}

void TLSSessionExtensionInspector::handleFetchResponse(const IpcMsgData& ipcMsg)
{
    if (!m_requestActive || m_requestKind != RequestKind::Entries)
    {
        return;
    }

    const auto& response = ipcMsg.m_data.m_tlsSessionExtensionTableFetchResponse;
    if (response.m_requestId != m_requestId || !isSelectedCore(response.m_coreId))
    {
        return;
    }

    auto& page = m_pageState[response.m_coreId];
    if (page.complete || response.m_numOfValidEntries > page.remaining)
    {
        return;
    }

    for (uint32_t index = 0;
         index < response.m_numOfValidEntries && index < MAX_TLS_SESSION_EXTENSION_RESULT_PER_ENGINE;
         ++index)
    {
        const auto& record = response.m_records[index];
        m_entryRows.push_back({response.m_coreId, record.m_sessionIndex, {
            std::to_string(response.m_coreId),
            std::to_string(record.m_sessionIndex),
            hsStateToString(record.m_hsState),
            tlsVersionMaskToString(record.m_tlsVersionMask),
            violationToString(record.m_terminalViolation)}});
    }

    page.remaining -= response.m_numOfValidEntries;
    if (!m_allSessions || response.m_isDone || page.remaining == 0 ||
        response.m_numOfValidEntries == 0 || !response.m_hasNextCursor)
    {
        completeEngine(response.m_coreId);
        return;
    }

    page.hasCursor = true;
    page.cursor = response.m_nextCursor;
    if (!sendFetchRequest(response.m_coreId, page.hasCursor, page.cursor, page.remaining))
    {
        completeEngine(response.m_coreId);
    }
}

void TLSSessionExtensionInspector::fillSizeResultStream(std::ostream* output, bool clearRows)
{
    TablePrinter::TableSorterByColumn sorter{compareEngineIds, 0};
    sorter.sortTable(m_sizeTablePrinter);
    m_sizeTablePrinter.printTable(output);
    if (clearRows)
    {
        m_sizeTablePrinter.clearTableRows();
    }
}

void TLSSessionExtensionInspector::fillEntryResultStream(std::ostream* output, bool clearRows)
{
    std::sort(m_entryRows.begin(), m_entryRows.end(), [](const EntryRow& first, const EntryRow& second) {
        if (first.engineId != second.engineId)
        {
            return first.engineId < second.engineId;
        }
        return first.sessionIndex < second.sessionIndex;
    });

    m_entryTablePrinter.clearTableRows();
    for (const auto& row : m_entryRows)
    {
        m_entryTablePrinter.addTableRow(row.cells);
    }

    TablePrinter::TableSorterByColumn sorter{compareEngineIds, 0};
    sorter.sortTable(m_entryTablePrinter);
    m_entryTablePrinter.printTable(output);
    if (clearRows)
    {
        m_entryTablePrinter.clearTableRows();
        m_entryRows.clear();
    }
}

bool TLSSessionExtensionInspector::hasEntryResults() const
{
    return !m_entryRows.empty();
}

STATUS TLSSessionExtensionInspectorObserver::init()
{
    if (m_ipcMessage == nullptr || m_visibility == nullptr)
    {
        return ERROR;
    }

    if (m_ipcMessage->registerObserver(
            IPC_MSG_TYPE_TLS_SESSION_EXTENSION_TABLE_SIZE_REQUEST, this) != OK)
    {
        return ERROR;
    }

    if (m_ipcMessage->registerObserver(
            IPC_MSG_TYPE_TLS_SESSION_EXTENSION_TABLE_FETCH_REQUEST, this) != OK)
    {
        return ERROR;
    }

    return OK;
}

void TLSSessionExtensionInspectorObserver::handleMsg(IpcMsgData& ipcMsg)
{
    switch (ipcMsg.m_msgType)
    {
        case IPC_MSG_TYPE_TLS_SESSION_EXTENSION_TABLE_SIZE_REQUEST:
            handleSizeRequest(ipcMsg);
            break;
        case IPC_MSG_TYPE_TLS_SESSION_EXTENSION_TABLE_FETCH_REQUEST:
            handleFetchRequest(ipcMsg);
            break;
        default:
            break;
    }
}

void TLSSessionExtensionInspectorObserver::handleSizeRequest(const IpcMsgData& ipcMsg)
{
    const auto& request = ipcMsg.m_data.m_tlsSessionExtensionTableSizeRequest;
    if (request.m_coreId != 0 && request.m_coreId != m_coreId)
    {
        return;
    }

    IpcMsgData response{};
    response.m_destMask = 0;
    response.m_msgType = IPC_MSG_TYPE_TLS_SESSION_EXTENSION_TABLE_SIZE_RESPONSE;
    response.m_data.m_tlsSessionExtensionTableSizeResponse.m_coreId = m_coreId;
    response.m_data.m_tlsSessionExtensionTableSizeResponse.m_size =
        m_visibility->getSessionExtensionCount();
    response.m_data.m_tlsSessionExtensionTableSizeResponse.m_requestId = request.m_requestId;
    m_ipcMessage->sendMsg(response);
}

void TLSSessionExtensionInspectorObserver::handleFetchRequest(const IpcMsgData& ipcMsg)
{
    const auto& request = ipcMsg.m_data.m_tlsSessionExtensionTableFetchRequest;
    if (request.m_coreId != 0 && request.m_coreId != m_coreId)
    {
        return;
    }

    IpcMsgData response{};
    response.m_destMask = 0;
    response.m_msgType = IPC_MSG_TYPE_TLS_SESSION_EXTENSION_TABLE_FETCH_RESPONSE;
    response.m_data.m_tlsSessionExtensionTableFetchResponse.m_coreId = m_coreId;
    response.m_data.m_tlsSessionExtensionTableFetchResponse.m_requestId = request.m_requestId;
    response.m_data.m_tlsSessionExtensionTableFetchResponse.m_activeSize =
        m_visibility->getSessionExtensionCount();
    response.m_data.m_tlsSessionExtensionTableFetchResponse.m_isDone = true;

    if (request.m_maxEntries == 0)
    {
        m_ipcMessage->sendMsg(response);
        return;
    }

    TLSFlow::Extension::Engine::TLSSessionExtensionInspectorQuery query{};
    query.allSessions = request.m_allSessions;
    query.sessionIndex = request.m_sessionIndex;
    query.maxEntries = std::min<uint32_t>(
        request.m_maxEntries, MAX_TLS_SESSION_EXTENSION_RESULT_PER_ENGINE);
    query.hasCursor = request.m_hasCursor;
    query.cursor = request.m_cursor;

    bool done = true;
    bool hasNextCursor = false;
    uint32_t nextCursor = 0;
    const bool fetched = m_visibility->fetchSessionExtensions(
        query,
        [&response](const TLSFlow::Extension::Engine::TLSSessionExtensionInspectorSnapshot& snapshot) {
            auto& result = response.m_data.m_tlsSessionExtensionTableFetchResponse;
            if (result.m_numOfValidEntries >= MAX_TLS_SESSION_EXTENSION_RESULT_PER_ENGINE)
            {
                return false;
            }

            auto& record = result.m_records[result.m_numOfValidEntries++];
            record.m_sessionIndex = snapshot.sessionIndex;
            record.m_hsState = static_cast<uint8_t>(snapshot.hsState);
            record.m_tlsVersionMask = snapshot.tlsVersionMask;
            record.m_clientHelloCount = snapshot.clientHelloCount;
            record.m_earlyDataCount = snapshot.earlyDataCount;
            record.m_earlyDataOffered = snapshot.earlyDataOffered;
            record.m_preCcsSeen = snapshot.preCcsSeen;
            record.m_fingerprintHigh = snapshot.chFingerprint.value().high;
            record.m_fingerprintLow = snapshot.chFingerprint.value().low;
            record.m_terminalViolation = static_cast<uint8_t>(
                TLSFlow::highestTLSViolation(snapshot.terminalViolationMask));
            record.m_clientHelloTimeStamp = snapshot.clientHelloTimeStamp;
            return result.m_numOfValidEntries < MAX_TLS_SESSION_EXTENSION_RESULT_PER_ENGINE;
        },
        done,
        hasNextCursor,
        nextCursor);

    if (fetched)
    {
        response.m_data.m_tlsSessionExtensionTableFetchResponse.m_isDone =
            done || response.m_data.m_tlsSessionExtensionTableFetchResponse.m_numOfValidEntries >=
                        request.m_maxEntries;
        response.m_data.m_tlsSessionExtensionTableFetchResponse.m_hasNextCursor =
            !response.m_data.m_tlsSessionExtensionTableFetchResponse.m_isDone && hasNextCursor;
        response.m_data.m_tlsSessionExtensionTableFetchResponse.m_nextCursor = nextCursor;
    }

    m_ipcMessage->sendMsg(response);
}

            {
            public:
                virtual ~ITLSSessionExtensionInspector() = default;

                virtual uint32_t getSessionExtensionCount() const = 0;

                virtual bool fetchSessionExtensions(
                    const TLSSessionExtensionInspectorQuery& query,
                    const TLSSessionExtensionInspectorVisitor& visitor,
                    bool& done,
                    bool& hasNextCursor,
                    uint32_t& nextCursor) = 0;
            };
        }

#pragma once

#include <array>

#include "../../common/include/TLSData.hpp"
#include "../../packet-flow/include/ITlsSubject.hpp"
#include "../../packet-flow/include/TLSPriority.hpp"

namespace TLSFlow
{
    namespace Extension
    {
        namespace Engine
        {
            /**
             * @brief Per-session TLS extension state.
             *
             * Owns the TLSData for a session and maintains per-event observer
             * lists, each with capacity equal to the number of priority slots
             * (TLSPriority::Max). Intended to be embedded in a session object
             * and recycled via init() / clear() rather than re-allocated.
             */
            class TLSSessionExtension
            {
            public:
                using Event       = PacketFlow::Engine::ITlsSubject::Event;
                using ListenerVec = ESTL::CTVector<
                    std::weak_ptr<PacketFlow::Engine::ITlsListener>,
                    static_cast<std::size_t>(PacketFlow::Engine::TLSPriority::Max),

#pragma once

#include <cstddef>
#include <cstdint>
#include <functional>

#include "../../common/include/Definitions.hpp"
#include "TLSSessionExtension.hpp"
#include <sh-data-struct/ESTL/FixedUnorderedMap.hpp>
#include <sh-data-struct/FIFOHashSet/FIFOSimpleSet.hpp>

namespace TLSFlow
{
    namespace Extension
    {
        namespace Engine
        {
            /**
             * @brief Per-session TLS extension storage table.
             *
             * Thread-safety: none. All calls must be serialised by the caller.
             */
            template <std::size_t MainCapacity, std::size_t ReclaimableCapacity>
            class TLSSessionExtensionTable
            {
                static_assert(MainCapacity >= 2,
                              "Main capacity must be at least 2");
                static_assert(ReclaimableCapacity > 0,
                              "Reclaimable capacity must be greater than zero");

                static constexpr std::size_t PrimaryCapacity = MainCapacity / 2;
                static constexpr std::size_t OverflowCapacity = MainCapacity - PrimaryCapacity;

                using Storage = ESTL::CTUnorderedMap<int,
                                                     TLSSessionExtension,
                                                     PrimaryCapacity,
                                                     OverflowCapacity,
                                                     std::hash<int>,
                                                     false>;
                using ReclaimableSet = FIFOSimpleSet<int, ReclaimableCapacity>;

            public:
                using FetchCallback = std::function<bool(int, const TLSSessionExtension&)>;

                TLSSessionExtensionTable() = default;
                virtual ~TLSSessionExtensionTable() = default;

                TLSSessionExtensionTable(const TLSSessionExtensionTable&)            = delete;
                TLSSessionExtensionTable& operator=(const TLSSessionExtensionTable&) = delete;

                /**
                 * @brief Allocates and returns an extension for @p sessionId.
                 *
                 * If the table is full, the oldest reclaimable extension is
                 * removed before the new extension is inserted.
                 *
                 * @return A pointer to the extension, or nullptr if no storage
                 *         is available.
                 */
                virtual TLSSessionExtension* alloc(int sessionId)
                {
                    auto* existing = m_storage.find(sessionId);
                    if (existing != nullptr)
                    {
                        return existing;
                    }

                    auto* extension = insertClean(sessionId);
                    if (extension != nullptr)
                    {
                        return extension;
                    }

                    if (m_reclaimable.empty())
                    {
                        return nullptr;
                    }

                    auto oldest = m_reclaimable.begin();
                    if (oldest == m_reclaimable.end())
                    {
                        return nullptr;
                    }

                    const int victimSessionId = *oldest;
                    auto* victim = m_storage.find(victimSessionId);
                    if (victim == nullptr)
                    {
                        m_reclaimable.erase(victimSessionId);
                        return nullptr;
                    }

                    m_reclaimable.erase(victimSessionId);
                    victim->resetValue();
                    if (!m_storage.erase(victimSessionId))
                    {
                        return nullptr;
                    }

                    return insertClean(sessionId);
                }

                /**
                 * @brief Returns the extension for @p sessionId, or nullptr.
                 *
                 * Does not create an entry. Safe to call before alloc() or
                 * after release().
                 */
                TLSSessionExtension* get(int sessionId)
                {
                    return m_storage.find(sessionId);
                }

                /**
                 * @brief Releases the extension for @p sessionId.
                 *
                 * Removes the entry from the table so that a future alloc()
                 * for the same ID returns a fresh, default-constructed
                 * extension.
                 *
                 * No-op if @p sessionId is not in the table.
                 */
                void release(int sessionId)
                {
                    m_reclaimable.erase(sessionId);

                    auto* extension = m_storage.find(sessionId);
                    if (extension != nullptr)
                    {
                        extension->resetValue();
                        m_storage.erase(sessionId);
                    }
                }

                virtual bool markReclaimable(int sessionId)
                {
                    if (m_storage.find(sessionId) == nullptr)
                    {
                        return false;
                    }

                    if (m_reclaimable.contains(sessionId))
                    {
                        return true;
                    }

                    return m_reclaimable.insert(sessionId);
                }

                uint32_t size() const
                {
                    return static_cast<uint32_t>(m_storage.size());
                }

                bool fetch(bool allSessions,
                           int sessionId,
                           uint32_t maxEntries,
                           bool hasCursor,
                           uint32_t cursor,
                           const FetchCallback& callback,
                           bool& done,
                           bool& hasNextCursor,
                           uint32_t& nextCursor)
                {
                    done = true;
                    hasNextCursor = false;
                    nextCursor = 0;

                    if (maxEntries == 0 || !callback)
                    {
                        return false;
                    }

                    if (!allSessions)
                    {
                        auto* extension = m_storage.find(sessionId);
                        if (extension != nullptr)
                        {
                            callback(sessionId, *extension);
                        }
                        return true;
                    }

                    const uint32_t tableCursor = hasCursor ? cursor : UINT32_MAX;
                    auto iterator = m_storage.beginFromCursor(tableCursor);
                    const auto end = m_storage.end();
                    uint32_t fetchedEntries = 0;
                    bool continueFetching = true;

                    while (iterator != end && fetchedEntries < maxEntries && continueFetching)
                    {
                        auto entry = *iterator;
                        continueFetching = callback(entry.first, entry.second);
                        ++iterator;
                        ++fetchedEntries;
                    }

                    if (iterator != end)
                    {
                        hasNextCursor = true;
                        done = false;
                        nextCursor = iterator.cursor();
                    }
                    return true;
                }

            private:
                TLSSessionExtension* insertClean(int sessionId)
                {
                    if (m_storage.size() >= MainCapacity)
                    {
                        return nullptr;
                    }

                    return m_storage.insertDefault(sessionId);
                }

                Storage m_storage;
                ReclaimableSet m_reclaimable;
            };

        } // namespace Engine
    } // namespace Extension
} // namespace TLSFlow

Misc::TLSVersion convertToMiscTlsVersion(uint16_t version) {
    switch (version) {
        case 0x0300:
            return Misc::TLSVersion::SSL_3_0;
        case 0x0301:
            return Misc::TLSVersion::V1_0;
        case 0x0302:
            return Misc::TLSVersion::V1_1;
        case 0x0303:
            return Misc::TLSVersion::V1_2;
        case 0x0304:
            return Misc::TLSVersion::V1_3;
        default:
            return Misc::TLSVersion::Other;
    }

int TLSFlowConductor::onEvent(Session * session, DPFrame * dpFrame, int event) noexcept
{
    switch (event) {
        case STM_EVENT_NEW_SESSION:
            return onNewSession(session, dpFrame);
        case STM_EVENT_INBOUND_PACKET_PROCESS:
            return onInbound(session, dpFrame);
        case STM_EVENT_AGING:
            return onAging(session);
        default:
            return RET_OBSERVER_CONTINUE;
    }
}

ObserverResult TLSFlowConductor::onNewSession(Session *session, DPFrame * /*packet*/)
{
    Extension* extension = buildExtension(*session);
    if (extension == nullptr) {
        return RET_OBSERVER_CONTINUE;
    }

    registerSessionObservers(*session);
    return RET_OBSERVER_CONTINUE;
}

ObserverResult TLSFlowConductor::onInbound(Session * session, DPFrame * packet)
{
    
    Extension* extension = nullptr;
    if (prepareInboundExtension(*session, *packet, extension)) {
        return RET_OBSERVER_CONTINUE;
    }

    if (extension == nullptr) {
        return RET_OBSERVER_CONTINUE;
    }

    return process(*session, *packet, *extension);
}

bool TLSFlowConductor::prepareInboundExtension(Session& session,
                                               DPFrame& packet,
                                               Extension*& extension)
{
    extension = m_extensionTable->get(session.getSessionIndex());

    if (packet.getPayloadLayer4Protocol() != PARSE_IP_PROTOCOL_TCP) {
        return false;
    }

    auto* tcpHeader = reinterpret_cast<IPTCPG_tcp_hdr_STC*>(
        packet.getPayloadLayer4Ptr());
    if (!IPTCPG_is_syn_no_ack_MAC(tcpHeader)) {
        return false;
    }

    if (extension == nullptr) {
        buildExtension(session);
        return true;
    }

    const auto& tlsData = extension->getTlsData();
    const bool hasRecordedViolation =
        tlsData.terminalViolationMask != TLSFlow::TLSViolationMaskNone;

    if (isActiveHandshakeState(tlsData.hsState) && !hasRecordedViolation) {
        return false;
    }

    m_extensionTable->release(session.getSessionIndex());
    buildExtension(session);
    extension = nullptr;
    return true;
}

ObserverResult TLSFlowConductor::process(Session& session, DPFrame& packet, Extension& extension)
{
    auto &tlsData = extension.getTlsData();

    if (packet.getPayloadLayer4Protocol() == PARSE_IP_PROTOCOL_TCP) {
        auto pTcpHeader = (IPTCPG_tcp_hdr_STC *)packet.getPayloadLayer4Ptr();
        if (SYN_PROTECT_IS_LEGAL_RST(pTcpHeader) && !SYN_PROTECT_IS_LEGAL_ACK(pTcpHeader)) {
            const auto verdict = notify(extension.getObserverList(Event::TCPReset), [&session, &packet, &tlsData](std::shared_ptr<ITlsListener> listener) {
                return listener->onTCPReset(session, packet, tlsData);
            });
            
            if (verdict.violationMask != TLSFlow::TLSViolationMaskNone) {
                if (verdict.dispatch == TLSFeatureVerdict::Dispatch::Stop) {
                    commitError(session, extension, verdict.violationMask);
                } else {
                    commitViolation(extension, verdict.violationMask);
                }
            } else {
                // Normal reset path: clear transient TLS data.
                tlsData.clear();
            }
            return toObserverResult(verdict);
        }
    }

    if (isSilentTerminalState(tlsData.hsState))
    {
        return RET_OBSERVER_CONTINUE;
    }

    uint16_t len = packet.getFrameTotalLen() - (packet.getPayloadLayer7Ptr() - packet.getLayer2Ptr());
    if (0 == len) {
        return RET_OBSERVER_CONTINUE;
    }


        if (usesTerminalPayloadPath(tlsData.hsState)) {
            return dispatchTerminalPayload(session, packet, extension);
        }

    uint32_t seq = packet.getTcpSeqNum();

    processor->beginPacket({packet.getPayloadLayer7Ptr(), len, static_cast<uint32_t>(session.getSessionIndex()), seq });

    IProcessor::TLSStepResult stepResult;
    do {
        stepResult = processor->processRecord(tlsData);

        const ObserverResult observerResult = dispatchStepResult(session, packet, extension, stepResult);
        if (observerResult == RET_OBSERVER_STOP) {
            return RET_OBSERVER_STOP;
        }
    } while (stepResult == IProcessor::TLSStepResult::RecordParsed &&
             isActiveHandshakeState(extension.getTlsData().hsState));

    return RET_OBSERVER_CONTINUE;
}

ObserverResult TLSFlowConductor::dispatchStepResult(Session &session, DPFrame &packet, Extension& extension, IProcessor::TLSStepResult stepResult)
{
    switch (stepResult) {
        case IProcessor::TLSStepResult::RecordParsed:
            return analyzeProcessResult(session, packet, extension);
        case IProcessor::TLSStepResult::Stop:
        {
            const auto verdict = checkForParsingError(session, packet, extension);
            if (verdict.violationMask != TLSFlow::TLSViolationMaskNone) {
                if (verdict.dispatch == TLSFeatureVerdict::Dispatch::Stop) {
                    commitError(session, extension, verdict.violationMask);
                } else {
                    commitViolation(extension, verdict.violationMask);
                }
            }
            return toObserverResult(verdict);
        }
        case IProcessor::TLSStepResult::NeedMoreData:
            return RET_OBSERVER_CONTINUE;
        case IProcessor::TLSStepResult::Unsupported:
            commitUnsupported(extension, processor->isContinuationGap());
            return RET_OBSERVER_CONTINUE;
        case IProcessor::TLSStepResult::EndOfPacket:
        default:
            return RET_OBSERVER_CONTINUE;
    }
}

ObserverResult TLSFlowConductor::analyzeProcessResult(Session &session, DPFrame &packet, Extension& extension)
{
    bool sawStop = false;
    TLSFlow::TLSViolationMask accumulatedViolationMask = TLSFlow::TLSViolationMaskNone;

    const auto applyVerdict = [&](const TLSFeatureVerdict& verdict) -> ObserverResult {
        if (verdict.violationMask != TLSFlow::TLSViolationMaskNone) {
            accumulatedViolationMask |= verdict.violationMask;
        }

        if (verdict.dispatch == TLSFeatureVerdict::Dispatch::Stop) {
            if (accumulatedViolationMask != TLSFlow::TLSViolationMaskNone) {
                commitError(session, extension, accumulatedViolationMask);
                return RET_OBSERVER_STOP;
            }
            sawStop = true;
        }
        return RET_OBSERVER_CONTINUE;
    };

    // 1) parsing error observers
    const auto parseVerdict = checkForParsingError(session, packet, extension);
    if (applyVerdict(parseVerdict) == RET_OBSERVER_STOP) {
        return RET_OBSERVER_STOP;
    }

    const auto& record = recordsDataViewer->getLastRecord();
    TLSRecordContext recordContext{
        record,
        record.isClientHello() ? clientHelloDataViewer->getSupportedVersionBitMask() : 0u,
        record.isClientHello() ? clientHelloDataViewer->isEarlyDataOffered() : false};
    if (record.isAlert() && record.readableAlert) {
        recordContext.alertTerminating = TLSFlow::isTerminatingAlert(
            extension.getTlsData().tlsVersionMask,
            record.alertLevel,
            record.alertDescription);
    }

    // 2) record observers
    const auto recordVerdict = onRecord(session, packet, extension, recordContext);
    if (applyVerdict(recordVerdict) == RET_OBSERVER_STOP) {
        return RET_OBSERVER_STOP;
    }

    // 3) client hello observers (only if complete)
    if (clientHelloDataViewer->getClientHelloParsingStatus() == IClientHelloDataViewer::ClientHelloParsingStatus::Complete) {
        const auto chVerdict = onClientHello(session, packet, extension);
        if (applyVerdict(chVerdict) == RET_OBSERVER_STOP) {
            return RET_OBSERVER_STOP;
        }
    }

    if (accumulatedViolationMask != TLSFlow::TLSViolationMaskNone) {
        if (sawStop) {
            commitError(session, extension, accumulatedViolationMask);
        } else {
            commitViolation(extension, accumulatedViolationMask);
        }

        return sawStop ? RET_OBSERVER_STOP : RET_OBSERVER_CONTINUE;
    }

    const auto previousHandshakeState = extension.getTlsData().hsState;
    commitNormalCurrentRecord(extension, recordContext, session.getLastPacketTimestamp());
    if (m_extensionTable &&
        previousHandshakeState != HsState::Finished &&
        extension.getTlsData().hsState == HsState::Finished) {
        m_extensionTable->markReclaimable(session.getSessionIndex());
    }

    return sawStop ? RET_OBSERVER_STOP : RET_OBSERVER_CONTINUE;
}

TLSFeatureVerdict TLSFlowConductor::checkForParsingError(Session &session, DPFrame &packet, Extension& extension)
{
    const ParsingError error = parsingErrorViewer->getLastError();
    if (error == ParsingError::None) {
        return TLSFeatureVerdict{};
    }

    const auto& observers = extension.getObserverList(Event::Anomaly);
    if (observers.empty()) {
        return TLSFeatureVerdict{};
    }

    auto& tlsData = extension.getTlsData();
    return notify(observers, [&session, &packet, &error, &tlsData](std::shared_ptr<ITlsListener> listener) {
        return listener->onParsingError(session, packet, error, tlsData);
    });
}

TLSFeatureVerdict TLSFlowConductor::onClientHello(Session &session, DPFrame &packet, Extension& extension)
{
    const auto& chObservers = extension.getObserverList(Event::ClientHello);
    if (chObservers.empty()) {
        return TLSFeatureVerdict{};
    }

    auto& tlsData = extension.getTlsData();
    Sni sni;
    clientHelloDataViewer->getSni(sni);
    Fingerprint fingerprint;
    clientHelloDataViewer->getFingerprint(fingerprint);

    ClientHelloData chData = {fingerprint,
                            sni,
                            convertToMiscTlsVersion(clientHelloDataViewer->getTlsVersion()),
                            clientHelloDataViewer->getCipherCount(),
                            clientHelloDataViewer->getExtensionCount(),
                            clientHelloDataViewer->getCompressionCount()};

    return notify(chObservers, [&session, &packet, &chData, &tlsData](std::shared_ptr<ITlsListener> listener) {
        return listener->onClientHello(session, packet, chData, tlsData);
    });
}

TLSFeatureVerdict TLSFlowConductor::onRecord(Session &session, DPFrame &packet, Extension& extension, const TLSRecordContext& context)
{
    const auto& observers = extension.getObserverList(Event::TLSRecord);
    if (observers.empty()) {
        return TLSFeatureVerdict{};
    }
    auto& tlsData = extension.getTlsData();

    return notify(observers, [&session, &packet, &context, &tlsData](std::shared_ptr<ITlsListener> listener) {
        return listener->onTLSRecord(session, packet, context, tlsData);
    });
}

ObserverResult TLSFlowConductor::dispatchTerminalPayload(Session& session,
                                                         DPFrame& packet,
                                                         Extension& extension)
{
    const auto& observers = extension.getObserverList(Event::TerminalPayload);
    if (observers.empty()) {
        return RET_OBSERVER_CONTINUE;
    }

    auto& tlsData = extension.getTlsData();

    const auto verdict = notify(observers, [&session, &packet, &tlsData](std::shared_ptr<ITlsListener> listener) {
        return listener->onTerminalPayload(session, packet, tlsData);
    });

    if (verdict.violationMask != TLSFlow::TLSViolationMaskNone) {
        if (verdict.dispatch == TLSFeatureVerdict::Dispatch::Stop) {
            commitError(session, extension, verdict.violationMask);
        } else {
            commitViolation(extension, verdict.violationMask);
        }
    }

    return toObserverResult(verdict);
}

ObserverResult TLSFlowConductor::onAging(Session * session)
{
    Extension* extension = m_extensionTable->get(session->getSessionIndex());
    if (extension == nullptr) {
        return RET_OBSERVER_CONTINUE;
    }

    auto& tlsData = extension->getTlsData();
    notify(extension->getObserverList(Event::SessionEnd), [&session, &tlsData](std::shared_ptr<ITlsListener> listener) {
        return listener->onSessionEnd(*session, tlsData);
    });
    m_extensionTable->release(session->getSessionIndex());
    return RET_OBSERVER_CONTINUE;
}

void TLSFlowConductor::subscribe(TLSPriority priority, std::weak_ptr<ITlsListener> listener)
{
    m_subscribers[priority] = listener;
}

void TLSFlowConductor::unsubscribe(TLSPriority priority)
{
    m_subscribers.erase(priority);
}

uint32_t TLSFlowConductor::getSessionExtensionCount() const
{
    return m_extensionTable == nullptr ? 0 : m_extensionTable->size();
}

bool TLSFlowConductor::fetchSessionExtensions(
    const TLSFlow::Extension::Engine::TLSSessionExtensionInspectorQuery& query,
    const TLSFlow::Extension::Engine::TLSSessionExtensionInspectorVisitor& visitor,
    bool& done,
    bool& hasNextCursor,
    uint32_t& nextCursor)
{
    if (m_extensionTable == nullptr || !visitor)
    {
        done = true;
        hasNextCursor = false;
        nextCursor = 0;
        return false;
    }

    return m_extensionTable->fetch(
        query.allSessions,
        query.sessionIndex,
        query.maxEntries,
        query.hasCursor,
        query.cursor,
        [&visitor](int sessionIndex, const Extension& extension) {
            const auto& tlsData = extension.getTlsData();
            TLSFlow::Extension::Engine::TLSSessionExtensionInspectorSnapshot snapshot{};
            snapshot.sessionIndex = sessionIndex;
            snapshot.hsState = tlsData.hsState;
            snapshot.tlsVersionMask = tlsData.tlsVersionMask;
            snapshot.clientHelloCount = tlsData.clientHelloCount;
            snapshot.earlyDataCount = tlsData.earlyDataCount;
            snapshot.earlyDataOffered = tlsData.earlyDataOffered;
            snapshot.preCcsSeen = tlsData.preCcsSeen;
            snapshot.chFingerprint = tlsData.chFingerprint;
            snapshot.terminalViolationMask = tlsData.terminalViolationMask;
            snapshot.clientHelloTimeStamp = tlsData.clientHelloTimeStamp;
            return visitor(snapshot);
        },
        done,
        hasNextCursor,
        nextCursor);
}

TLSFlowConductor::Extension* TLSFlowConductor::buildExtension(Session& session)
{
    Extension* ext = nullptr;

    for (auto& entry : m_subscribers) {
        auto sharedPtr = entry.second.lock();
        if (!sharedPtr) {
            continue;
        }

        ITlsListener::EventMask mask = sharedPtr->getSessionEventMask(session);
        if (mask == 0) {
            continue;
        }

        if (ext == nullptr) {
            if (!m_extensionTable) {
                return nullptr;
            }
            ext = m_extensionTable->alloc(session.getSessionIndex());
            if (ext == nullptr) {
                return nullptr;
            }
        }

        addListenerToExtension(*ext, sharedPtr, mask);
    }

    return ext;
}

void TLSFlowConductor::registerSessionObservers(Session& session)
{
    session.registerObserver(this, STM_EVENT_INBOUND_PACKET_PROCESS, InboundPriorityTLS);
    session.registerObserver(this, STM_EVENT_AGING, AgingPriorityTLS);
}

void TLSFlowConductor::addListenerToExtension(Extension& extension,
                                              const std::shared_ptr<ITlsListener>& listener,
                                              ITlsListener::EventMask mask)
{
    for (uint8_t i = 0; i < static_cast<uint8_t>(Event::MaxTypes); ++i) {
        if (mask & static_cast<ITlsListener::EventMask>(1u << i)) {
            extension.addObserver(static_cast<Event>(i), listener);
        }
    }
}

void TLSFlowConductor::commitNormalCurrentRecord(Extension &extension,
                                                 const TLSRecordContext& context,
                                                 unsigned long packetTimestamp)
{
    auto& tls = extension.getTlsData();
    if (usesTerminalPayloadPath(tls.hsState)) {
        return;
    }

    switch(context.record.recordType) {
        case static_cast<uint8_t>(RecordTypesE::handshakeContent):
            if (tls.hsState == HsState::ChangeCipherSpec) {
                tls.hsState = HsState::Finished;
            } else if(context.record.isClientHello()) {
                updateTLSDataOnClientHello(tls, context, packetTimestamp);
            }
            break;
        case static_cast<uint8_t>(RecordTypesE::changeCipherSpecContent):
            updateTLSDataOnChangeCipherSpec(tls);
            break;
        case static_cast<uint8_t>(RecordTypesE::applicationDataContent):
            updateTLSDataOnApplicationData(tls);
            break;
        case static_cast<uint8_t>(RecordTypesE::alertContent):
            updateTLSDataOnAlert(tls, context);
            break;
            
        default:
            break;
    }
}

void TLSFlowConductor::updateTLSDataOnClientHello(TLSFlow::TLSData& tls,
                                                  const TLSRecordContext& context,
                                                  unsigned long packetTimestamp)
{
    if (clientHelloDataViewer->getClientHelloParsingStatus() !=
        NSProcessor::Engine::IClientHelloDataViewer::ClientHelloParsingStatus::Complete) {
        return;
    }

    ++tls.clientHelloCount;
    tls.clientHelloTimeStamp = packetTimestamp;
    clientHelloDataViewer->getFingerprint(tls.chFingerprint);
    tls.tlsVersionMask = clientHelloDataViewer->getSupportedVersionBitMask();
    tls.earlyDataOffered = context.earlyDataOffered;

    if (tls.hsState == HsState::PreHandshake) {
        tls.hsState = HsState::InProgress;
    }
}

void TLSFlowConductor::updateTLSDataOnChangeCipherSpec(TLSFlow::TLSData& tls)
{
    if(tls.clientHelloCount == 0) {
        ++tls.preCcsSeen;
    } else if (!TLSFlow::isTLS13(tls.tlsVersionMask) && tls.hsState == HsState::InProgress) {
        tls.hsState = HsState::ChangeCipherSpec;
    }
}

void TLSFlowConductor::updateTLSDataOnApplicationData(TLSFlow::TLSData& tls)
{
    if (tls.clientHelloCount == 0) {
        ++tls.earlyDataCount;
    } else if (TLSFlow::isTLS13(tls.tlsVersionMask) &&
               tls.hsState == HsState::InProgress &&
               !tls.earlyDataOffered) {
        tls.hsState = HsState::Finished;
    }
}

void TLSFlowConductor::updateTLSDataOnAlert(TLSFlow::TLSData& tls,
                                           const TLSRecordContext& context)
{
    if (!isActiveHandshakeState(tls.hsState) ||
        !context.record.isAlert() ||
        !context.record.readableAlert ||
        !context.alertTerminating) {
        return;
    }

    tls.hsState = HsState::TerminatedByAlert;
}

void TLSFlowConductor::commitUnsupported(Extension &extension, bool continuationGap)
{
    auto& tls = extension.getTlsData();
    if (continuationGap ||
        tls.hsState == HsState::PreHandshake ||
        (tls.hsState == HsState::InProgress && clientHelloDataViewer &&
         clientHelloDataViewer->getClientHelloParsingStatus() ==
             IClientHelloDataViewer::ClientHelloParsingStatus::Partial))
    {
        tls.hsState = HsState::Unsupported;
    }
}

void TLSFlowConductor::commitViolation(Extension &extension, TLSFlow::TLSViolationMask violationMask)
{
    extension.getTlsData().terminalViolationMask |= violationMask;
}

void TLSFlowConductor::commitError(Session &session, Extension &extension, TLSFlow::TLSViolationMask violationMask)
{
    auto& dst = extension.getTlsData();
    dst.hsState = HsState::Error;
    commitViolation(extension, violationMask);
}

ObserverResult TLSFlowConductor::toObserverResult(const TLSFeatureVerdict& v)
{
    return (v.dispatch == TLSFeatureVerdict::Dispatch::Stop) ? RET_OBSERVER_STOP : RET_OBSERVER_CONTINUE;
}

            class ExtensionLocator {
            public:
                ExtensionLocator() = default;

                ExtensionLocator(ExtensionTypeE i_type, uint16_t i_length, const uint8_t *i_pValue) {
                    type = i_type;
                    length = i_length;
                    pValue = i_pValue;
                }

                ExtensionLocator(uint16_t i_type, uint16_t i_length, const uint8_t *i_pValue) {
                    type = static_cast<ExtensionTypeE>(i_type);
                    length = i_length;
                    pValue = i_pValue;
                }

                void setLengthFirstByte(uint8_t byte) { *(reinterpret_cast<uint8_t*>(&length))  = byte;}
                void setLengthLastByte(uint8_t byte) { *(reinterpret_cast<uint8_t*>(&length) + 1)  = byte;}
                void setTypeLastByte(uint8_t byte) {
                    uint16_t t = to_underlying(type) | byte;
                    type = static_cast<ExtensionTypeE>(t);
                }
                ExtensionTypeE getType() const { return type; }
                void setType(ExtensionTypeE inType) { type = inType; }
                void swapLength() {auto t = length; length = htons(t); }
                uint16_t getLength() const { return length; }
                void setLength(uint16_t inLength) { length = inLength; }
                const uint8_t *getPValue() const { return pValue; }
                void setPValue(const uint8_t *pValueIn) { pValue = pValueIn; }


                ExtensionTypeE type;
                uint16_t length;
                const uint8_t *pValue;

                /* Extensions Sorting function. The order is given by the following:
             * Extension value within compared objects are deGREASEd already (meaning only 0x0a0a is expected)

                1.	Ascending Extension Type value
                2.	Ascending Extension length
                3.	Ascending Extension value
                4.  Greased Extensions last
             */

                static bool extensionsSortingComparison(const ExtensionLocator &a, const ExtensionLocator &b) {
                    //cout << "extensionsSortingComparison START \n\ta.type  ( " << hex << showbase << internal << setfill('0') << setw(6) << static_cast<uint16_t >(a.type) << " ) " << "  \tb.type ( "  << setw(6) << static_cast<uint16_t >(b.type) << " ) " << endl;
                    if (a.type == ExtensionTypeE::NORMALIZED_GREASE) {
                        //cout << "a is grease" << endl;
                        if (b.type == ExtensionTypeE::NORMALIZED_GREASE) {
                            //cout << "b is also grease" << endl;
                            return false;
                        }
                        //cout << " ExtensionTypeE::NORMALIZED_GREASE < b.type ==" << ((ExtensionTypeE::NORMALIZED_GREASE < b.type) ? " true " : " false ") << endl;
                        return ExtensionTypeE::NORMALIZED_GREASE < b.type;
                    } else if (b.type == ExtensionTypeE::NORMALIZED_GREASE) {
                        //cout << "b is grease" << endl;
                        return a.type < ExtensionTypeE::NORMALIZED_GREASE;
                    }
                    if (a.type != b.type) {
                        //cout << "a.type != b.type, return a.type < b.type == " << ((a.type < b.type) ? "true" : "false") << endl;
                        return a.type < b.type;
                    }
                    if (a.length != b.length) {
                        //cout << "a.lenght != b.length, return a.length < b.length == " << ((a.length < b.length) ? "true" : "false") << endl;
                        return a.length < b.length;
                    }
                    return (memcmp(a.pValue, b.pValue, min(a.length, b.length)) < 0);
                }
            };

            class ExtensionsArray {
            public:
                ExtensionsArray() {}

                bool add(const ExtensionLocator &extension) {
                    if (maxSupportedExtensions <= m_numOfExtensions) {
                        return false;
                    }
                    m_array[m_numOfExtensions++] = extension;
                    return true;

                }

                /* ExtensionLocator - assumes input index is within valid range (get from getNumOfExtensions() )*/
                const ExtensionLocator &get(uint16_t index) const { return m_array[index]; }

                void reset() { m_numOfExtensions = 0; }

                uint16_t getNumOfExtensions() const { return m_numOfExtensions; }

                ExtensionLocator &getLast() { return m_array[getNumOfExtensions() == 0 ? getNumOfExtensions() : getNumOfExtensions() - 1]; }

                void sort() {
                    std::sort(m_array.begin(), m_array.begin() + getNumOfExtensions(),
                              TLSFlow::NSProcessor::Engine::ExtensionLocator::extensionsSortingComparison);
                }


            private:
                static constexpr uint32_t maxSupportedExtensions = 1024; // Maximal extensions supported to be parsed and sorted in a single Client Hello message
                uint16_t m_numOfExtensions;
                std::array<ExtensionLocator, maxSupportedExtensions> m_array;

            };
        }

    m_arrayIndex = 0;
    return true;
}
// NOTE! This API does not perform any calculation
// Save a portion of bytes to hashlater
bool Md5Digest::calculate(const uint8_t *const data, size_t len) {
    if (len == 0 ||
       (len + m_arrayIndex >= sizeof(m_array))) {
        return false;
    }
    memcpy(&(m_array[m_arrayIndex]), data, len);
    m_arrayIndex+= len;
    return true;
}

// Finalize iterative hash calculation (add padding if needed)
// This is actually the only place where we calculate the MD5 hash on the complete buffer aggregated until here
bool Md5Digest::finalize(TLSFlow::Fingerprint &fp){
    MD5(m_array, m_arrayIndex, (reinterpret_cast<uint8_t*>(&fp)));
    return true;
}

bool serialLess(uint32_t left, uint32_t right)
{
    return static_cast<int32_t>(left - right) < 0;
}

bool serialLessOrEqual(uint32_t left, uint32_t right)
{
    return left == right || serialLess(left, right);
}

bool serialGreater(uint32_t left, uint32_t right)
{
    return serialLess(right, left);
}

uint8_t tlsVersionMaskFromWireVersion(uint16_t version)
{
    using TLSFlow::TLSVersionBit;

    switch (version)
    {
        case 0x0300:
            return static_cast<uint8_t>(1u << static_cast<uint8_t>(TLSVersionBit::SSL30));
        case 0x0301:
            return static_cast<uint8_t>(1u << static_cast<uint8_t>(TLSVersionBit::TLS10));
        case 0x0302:
            return static_cast<uint8_t>(1u << static_cast<uint8_t>(TLSVersionBit::TLS11));
        case 0x0303:
            return static_cast<uint8_t>(1u << static_cast<uint8_t>(TLSVersionBit::TLS12));
        case 0x0304:
            return static_cast<uint8_t>(1u << static_cast<uint8_t>(TLSVersionBit::TLS13));
        default:
            return 0;
    }
}

void Processor::beginPacket(const PayloadParams &payloadParams)
{
    m_stepPacketError = false;
    m_stepPacketIsRetransmission = false;
    m_stepContinuationGap = false;
    m_stepContinuationMode = IProcessor::ContinuationMode::None;
    captureStepPacketMetadata(payloadParams);
    m_pSegmentBuffer = &m_initialSegment;
    m_pExtensionLocators = m_pSegmentBuffer->getLocatorsArray();
    m_pSegmentBuffer->setState(SegmentBuffer::State::InitialNonSegmented);
    m_pDigest = &m_pSegmentBuffer->getDigest();
    setState(ProcessorState::SingleSegment);

    // Always keep the live packet cursor available for the current packet.
    m_parserLocationPtr = m_packetPayloadPtr;
    m_lengthLeftToParse = m_packetPayloadLen;

    const bool resumed = resumeOrDiscardStepContinuation(payloadParams);
    if (resumed)
    {
        if (!m_stepPacketIsRetransmission && !m_stepContinuationGap)
        {
            m_records.clear();
            m_record = Record{};
        }
        return;
    }

    if (m_stepPacketError)
    {
        m_stepPacketInitialized = false;
        return;
    }

    resetStepState();
}

bool Processor::resumeOrDiscardStepContinuation(const PayloadParams &payloadParams)
{
    bool resumed = false;
    bool hasPendingBuffer = segmentBufferManager->access(payloadParams.getSessionIdx(), [&](auto& segmentBufferHand)
    {
        const uint32_t start = payloadParams.getTcpSeq();
        const uint32_t end = start + payloadParams.getLength();
        const uint32_t expected = segmentBufferHand.getNextSequence();

        if (start == expected)
        {
            resumed = resumeStepFromSegmentBuffer(payloadParams, segmentBufferHand, 0);
        }
        else if (serialLessOrEqual(end, expected))
        {
            m_stepPacketIsRetransmission = true;
            m_diagnosticsCounters->incEvent(DiagnosticsEvents::ContinuationFullRetransmission);
            resumed = true;
        }
        else if (serialLess(start, expected) && serialGreater(end, expected))
        {
            const uint32_t overlap = expected - start;
            m_parserLocationPtr += overlap;
            m_lengthLeftToParse -= static_cast<uint16_t>(overlap);
            m_diagnosticsCounters->incEvent(DiagnosticsEvents::ContinuationPartialOverlap);
            resumed = resumeStepFromSegmentBuffer(payloadParams, segmentBufferHand,
                                                  static_cast<uint16_t>(overlap));
        }
        else
        {
            m_stepContinuationGap = true;
            switch (segmentBufferHand.getContinuationMode())
            {
                case SegmentBuffer::ContinuationMode::OpaqueRecordTracking:
                    m_stepContinuationMode = IProcessor::ContinuationMode::OpaqueRecordTracking;
                    break;
                case SegmentBuffer::ContinuationMode::ClientHelloExtensionReassembly:
                    m_stepContinuationMode = IProcessor::ContinuationMode::ClientHelloExtensionReassembly;
                    break;
                case SegmentBuffer::ContinuationMode::AlertRecordReassembly:
                    m_stepContinuationMode = IProcessor::ContinuationMode::AlertRecordReassembly;
                    break;
                case SegmentBuffer::ContinuationMode::None:
                default:
                    m_stepContinuationMode = IProcessor::ContinuationMode::None;
                    break;
            }
            m_diagnosticsCounters->incEvent(DiagnosticsEvents::ContinuationGap);
            if (segmentBufferHand.getContinuationMode() == SegmentBuffer::ContinuationMode::OpaqueRecordTracking)
            {
                m_diagnosticsCounters->incEvent(DiagnosticsEvents::OpaqueRecordContinuationGap);
            }
            else if (segmentBufferHand.getContinuationMode() == SegmentBuffer::ContinuationMode::ClientHelloExtensionReassembly)
            {
                m_diagnosticsCounters->incEvent(DiagnosticsEvents::ClientHelloContinuationGap);
            }
        }
    });

    if (!hasPendingBuffer)
    {
        return false;
    }

    if (m_stepContinuationGap)
    {
        segmentBufferManager->deallocate(payloadParams.getSessionIdx());
        m_diagnosticsCounters->incEvent(DiagnosticsEvents::SegmentedStateEntriesFree);
        resetContinuationPointers();
        return true;
    }

    if (resumed)
    {
        return true;
    }

    segmentBufferManager->deallocate(payloadParams.getSessionIdx());
    resetContinuationPointers();
    return false;
}

IProcessor::TLSStepResult Processor::processRecord(const TLSData &data)
{
    if (!m_stepPacketInitialized)
    {
        return IProcessor::TLSStepResult::Stop;
    }

    if (m_stepContinuationGap)
    {
        return IProcessor::TLSStepResult::Unsupported;
    }

    if (m_lengthLeftToParse == 0)
    {
        return IProcessor::TLSStepResult::EndOfPacket;
    }

    if (m_stepPacketIsRetransmission)
    {
        return IProcessor::TLSStepResult::NeedMoreData;
    }


    // Delegate to the legacy single-record processing flow and translate the result.
    RecordProcessResult r = processSingleRecord(data);
    switch (r)
    {
        case RecordProcessResult::ContinueProcessing:
            return IProcessor::TLSStepResult::RecordParsed;
        case RecordProcessResult::Segmented:
            return IProcessor::TLSStepResult::NeedMoreData;
        case RecordProcessResult::Unsupported:
            return IProcessor::TLSStepResult::Unsupported;
        case RecordProcessResult::EndOfPacket:
            return IProcessor::TLSStepResult::EndOfPacket;
        case RecordProcessResult::Error:
        case RecordProcessResult::StopProcessing:
        default:
            return IProcessor::TLSStepResult::Stop;
    }
}

bool Processor::resumeStepFromSegmentBuffer(const PayloadParams& payloadParams,
                                             SegmentBuffer& segmentBuffer,
                                             uint16_t overlap)
{
    m_pSegmentBuffer = &segmentBuffer;
    m_pDigest = &m_pSegmentBuffer->getDigest();
    m_pExtensionLocators = m_pSegmentBuffer->getLocatorsArray();
    if (segmentBuffer.getContinuationMode() == SegmentBuffer::ContinuationMode::ClientHelloExtensionReassembly)
    {
        setState(ProcessorState::LastSegment);
        m_stepContinuationMode = IProcessor::ContinuationMode::ClientHelloExtensionReassembly;
    }
    else if (segmentBuffer.getContinuationMode() == SegmentBuffer::ContinuationMode::AlertRecordReassembly)
    {
        setState(ProcessorState::SingleSegment);
        m_stepContinuationMode = IProcessor::ContinuationMode::AlertRecordReassembly;
    }
    else if (segmentBuffer.getContinuationMode() == SegmentBuffer::ContinuationMode::OpaqueRecordTracking)
    {
        setState(ProcessorState::SingleSegment);
        m_stepContinuationMode = IProcessor::ContinuationMode::OpaqueRecordTracking;
    }
    else
    {
        setState(ProcessorState::SingleSegment);
        m_stepContinuationMode = IProcessor::ContinuationMode::None;
    }
    m_nextExpectedSeq = payloadParams.getTcpSeq() + payloadParams.getLength();
    if (segmentBuffer.getContinuationMode() == SegmentBuffer::ContinuationMode::ClientHelloExtensionReassembly &&
        payloadParams.getLength() > overlap)
    {
        segmentBuffer.setClientHelloSegmentCount(segmentBuffer.getClientHelloSegmentCount() + 1);
    }
    if (overlap != 0)
    {
        m_stepPacketIsRetransmission = false;
    }
    return true;
}

bool Processor::prepareSegmentBufferForContinuation(SegmentBuffer::ContinuationMode mode, bool &allocated)
{
    allocated = false;

    if (m_pSegmentBuffer != &m_initialSegment)
    {
        if (m_pSegmentBuffer->getContinuationMode() != SegmentBuffer::ContinuationMode::None)
        {
            return false;
        }

        m_pSegmentBuffer->resetContinuationState();
        m_pSegmentBuffer->setContinuationMode(mode);
        m_pExtensionLocators = m_pSegmentBuffer->getLocatorsArray();
        return true;
    }

    bool found = false;
    bool reused = false;
    segmentBufferManager->access(m_sessionIdx, [&](auto& segmentBufferHand)
    {
        found = true;
        if (segmentBufferHand.getContinuationMode() == SegmentBuffer::ContinuationMode::None)
        {
            segmentBufferHand.resetContinuationState();
            segmentBufferHand.setContinuationMode(mode);
            m_pSegmentBuffer = &segmentBufferHand;
            m_pExtensionLocators = m_pSegmentBuffer->getLocatorsArray();
            reused = true;
        }
    });

    if (reused)
    {
        return true;
    }

    if (found)
    {
        return false;
    }

    if (!segmentBufferManager->allocate(m_sessionIdx, [&](auto& segmentBufferHand)
    {
        segmentBufferHand.init();
        segmentBufferHand.setContinuationMode(mode);
        m_pSegmentBuffer = &segmentBufferHand;
        m_pExtensionLocators = m_pSegmentBuffer->getLocatorsArray();
    }))
    {
        return false;
    }

    allocated = true;
    m_diagnosticsCounters->incEvent(DiagnosticsEvents::SegmentedStateEntriesAlloc);
    return true;
}

void Processor::finishSegmentBufferContinuation()
{
    if (m_pSegmentBuffer == &m_initialSegment)
    {
        return;
    }

    if (m_pSegmentBuffer->getAlertAlignmentOffset() == 0)
    {
        deallocateSegmentBuff();
        return;
    }

    const uint8_t alignmentOffset = m_pSegmentBuffer->getAlertAlignmentOffset();
    m_pSegmentBuffer->resetContinuationState();
    m_pSegmentBuffer->setAlertAlignmentOffset(alignmentOffset);
    m_pSegmentBuffer->setNextSequence(m_nextExpectedSeq);
    m_pDigest = &m_pSegmentBuffer->getDigest();
    m_pExtensionLocators = m_pSegmentBuffer->getLocatorsArray();
    setState(ProcessorState::SingleSegment);
}

Processor::RecordProcessResult Processor::processSingleRecord(const TLSData &data)
{
    if (m_pSegmentBuffer->isAlertRecordReassembly())
    {
        return processSegmentedAlertContinuation();
    }

    if (m_pSegmentBuffer->getContinuationMode() == SegmentBuffer::ContinuationMode::OpaqueRecordTracking)
    {
        return processSegmentedOpaqueContinuation();
    }

    if (m_pSegmentBuffer->getContinuationMode() == SegmentBuffer::ContinuationMode::ClientHelloExtensionReassembly)
    {
        return processSegmentedClientHelloContinuation();
    }

    return processNewRecord(data);
}

Processor::RecordProcessResult Processor::processSegmentedClientHelloContinuation()
{
    const uint16_t extensionRemaining = m_pSegmentBuffer->getExtensionRemainingLength();
    uint16_t totalRecordLength = m_pSegmentBuffer->getTotalRecordLength();
    const uint16_t bytesToCopy = std::min(extensionRemaining, m_lengthLeftToParse);

    m_numOfCipherSuites = m_pSegmentBuffer->getNumOfCiphers();
    m_numOfCompressions = m_pSegmentBuffer->getNumOfCompressions();
    lastVersion = m_pSegmentBuffer->getLastVersion();
    m_clientHelloParsingStatus = IClientHelloDataViewer::ClientHelloParsingStatus::Partial;
    if (bytesToCopy != 0)
    {
        std::memcpy(m_pSegmentBuffer->getBuffer().data() + m_pSegmentBuffer->getBufferUsed(),
                    m_parserLocationPtr, bytesToCopy);
        m_pSegmentBuffer->setBufferUsed(m_pSegmentBuffer->getBufferUsed() + bytesToCopy);
        m_pSegmentBuffer->setExtensionRemainingLength(extensionRemaining - bytesToCopy);
        m_pSegmentBuffer->setContinuationRemainingLength(
            m_pSegmentBuffer->getContinuationRemainingLength() - bytesToCopy);
        m_parserLocationPtr += bytesToCopy;
        m_lengthLeftToParse -= bytesToCopy;
        m_pSegmentBuffer->setNextSequence(m_nextExpectedSeq);
    }

    if (m_pSegmentBuffer->getExtensionRemainingLength() != 0)
    {
        return RecordProcessResult::Segmented;
    }

    m_pSegmentBuffer->setNextLength(0);
    auto innerProcessResult = processExtensions();
    if (RecordProcessResult::ContinueProcessing != innerProcessResult)
    {
        if (RecordProcessResult::Error == innerProcessResult)
        {
            m_diagnosticsCounters->incEvent(DiagnosticsEvents::ExtensionParserFailures);
        }
        deallocateSegmentBuff();
        return innerProcessResult;
    }

    if (!m_supportedVersionsParsed)
    {
        m_supportedVersionsBitMask = tlsVersionMaskFromWireVersion(m_parsers->clientHello->getVersion());
    }

    const uint32_t clientHelloSegmentCount = m_pSegmentBuffer->getClientHelloSegmentCount();
    m_pDigest->finalize(m_fp);
    m_initialSegment.getDigest().copy(*m_pDigest);
    finishSegmentBufferContinuation();
    m_clientHelloParsingStatus = IClientHelloDataViewer::ClientHelloParsingStatus::Complete;

    // Add the ClientHello record to the records list
    m_record = Record{};
    m_record.recordType = static_cast<uint8_t>(RecordTypesE::handshakeContent);
    m_record.handshakeType = static_cast<uint8_t>(HandShakeTypesE::client_hello);
    m_record.recordLength = totalRecordLength;
    if(m_records.size() >= maxRecords){
        m_lastError = ParsingError::RecordsExceedDataViewerCapacity;
        return RecordProcessResult::StopProcessing;
    }
    m_records.push_back(m_record);

    if (clientHelloSegmentCount == 2)
    {
        m_diagnosticsCounters->incEvent(DiagnosticsEvents::TwoSegmentClientHelloCompleted);
    }
    else if (clientHelloSegmentCount >= 3)
    {
        m_diagnosticsCounters->incEvent(DiagnosticsEvents::ThreeOrMoreSegmentClientHelloCompleted);
    }

    // Reset state for processing any remaining records
    setState(ProcessorState::SingleSegment);

    return RecordProcessResult::ContinueProcessing;
}

Processor::RecordProcessResult Processor::processSegmentedOpaqueContinuation()
{
    const uint16_t remaining = m_pSegmentBuffer->getContinuationRemainingLength();
    const uint16_t consumed = std::min(remaining, m_lengthLeftToParse);

    m_parserLocationPtr += consumed;
    m_lengthLeftToParse -= consumed;
    m_pSegmentBuffer->setContinuationRemainingLength(remaining - consumed);
    m_pSegmentBuffer->setNextSequence(m_nextExpectedSeq);

    if (remaining > consumed)
    {
        return RecordProcessResult::Segmented;
    }

    m_record = Record{};
    m_record.recordType = m_pSegmentBuffer->getContinuationRecordType();
    m_record.handshakeType = m_pSegmentBuffer->getContinuationHandshakeType();
    m_record.recordLength = m_pSegmentBuffer->getContinuationRecordLength();
    if (m_records.size() >= maxRecords)
    {
        m_lastError = ParsingError::RecordsExceedDataViewerCapacity;
        finishSegmentBufferContinuation();
        return RecordProcessResult::StopProcessing;
    }
    m_records.push_back(m_record);
    finishSegmentBufferContinuation();
    return RecordProcessResult::ContinueProcessing;
}

bool Processor::startSegmentedAlert(const TLSData &data)
{
    bool allocated = false;
    if (!prepareSegmentBufferForContinuation(SegmentBuffer::ContinuationMode::AlertRecordReassembly, allocated))
    {
        return false;
    }

    auto& segmentBuffer = *m_pSegmentBuffer;
    const uint16_t bytesReceived = m_lengthLeftToParse;
    const uint16_t bytesToStore = std::min<uint16_t>(bytesReceived, TLSRecordHeaderSize + 2);
    if (bytesToStore != 0)
    {
        std::memcpy(segmentBuffer.getBuffer().data(), m_parserLocationPtr, bytesToStore);
    }
    segmentBuffer.setBufferUsed(bytesToStore);
    segmentBuffer.setNextSequence(m_nextExpectedSeq);
    segmentBuffer.setContinuationMode(SegmentBuffer::ContinuationMode::AlertRecordReassembly);
    segmentBuffer.setContinuationRecordType(static_cast<uint8_t>(RecordTypesE::alertContent));
    segmentBuffer.setContinuationHandshakeType(static_cast<uint8_t>(HandShakeTypesE::hello_request_RESERVED));
    segmentBuffer.setContinuationRecordLength(0);
    segmentBuffer.setContinuationRemainingLength(0);
    segmentBuffer.setAlertRecordReassembly(true);
    segmentBuffer.setAlertPlaintextVisible(
        data.hsState == TLSFlow::HsState::PreHandshake ||
        data.hsState == TLSFlow::HsState::InProgress);
    segmentBuffer.setAlertRecordBytesReceived(bytesReceived);
    segmentBuffer.setAlertHeaderParsed(false);
    m_stepContinuationMode = IProcessor::ContinuationMode::AlertRecordReassembly;
    return true;
}

Processor::RecordProcessResult Processor::processSegmentedAlertContinuation()
{
    auto& segmentBuffer = *m_pSegmentBuffer;

    while (true)
    {
        if (!segmentBuffer.isAlertHeaderParsed())
        {
            const uint16_t bytesReceived = segmentBuffer.getAlertRecordBytesReceived();
            const uint16_t bytesNeeded = TLSRecordHeaderSize - bytesReceived;
            const uint16_t bytesToConsume = std::min(bytesNeeded, m_lengthLeftToParse);
            if (bytesToConsume != 0)
            {
                std::memcpy(segmentBuffer.getBuffer().data() + bytesReceived,
                            m_parserLocationPtr,
                            bytesToConsume);
                m_parserLocationPtr += bytesToConsume;
                m_lengthLeftToParse -= bytesToConsume;
                segmentBuffer.setAlertRecordBytesReceived(bytesReceived + bytesToConsume);
                segmentBuffer.setBufferUsed(bytesReceived + bytesToConsume);
                segmentBuffer.setNextSequence(m_nextExpectedSeq);
            }

            const uint16_t updatedBytesReceived = segmentBuffer.getAlertRecordBytesReceived();
            if (updatedBytesReceived < TLSRecordHeaderSize)
            {
                segmentBuffer.setContinuationRemainingLength(TLSRecordHeaderSize - updatedBytesReceived);
                return RecordProcessResult::Segmented;
            }

            uint16_t consumed = 0;
            const uint8_t *header = reinterpret_cast<const uint8_t *>(segmentBuffer.getBuffer().data());
            if (m_parsers->tlsRecord->parse(header, TLSRecordHeaderSize, consumed) != ParsingStatus::Complete)
            {
                m_lastError = ParsingError::MalformedTlsRecord;
                finishSegmentBufferContinuation();
                return RecordProcessResult::Error;
            }

            if (!m_parsers->tlsRecord->isSupportedRecordType())
            {
                m_lastError = ParsingError::UnsupportedRecordType;
                finishSegmentBufferContinuation();
                return RecordProcessResult::StopProcessing;
            }

            const uint16_t recordPayloadLength = m_parsers->tlsRecord->getTlsRecordLength();
            if (recordPayloadLength > MaxTlsRecordPayloadSize)
            {
                m_lastError = ParsingError::RecordLengthExceedsData;
                finishSegmentBufferContinuation();
                return RecordProcessResult::Error;
            }
            if (m_clientHelloParsingStatus == IClientHelloDataViewer::ClientHelloParsingStatus::None)
            {
                lastVersion = m_parsers->tlsRecord->getTlsVersion();
            }

            const uint16_t totalRecordLength = TLSRecordHeaderSize + recordPayloadLength;
            if (updatedBytesReceived > totalRecordLength)
            {
                m_lastError = ParsingError::MalformedTlsRecord;
                finishSegmentBufferContinuation();
                return RecordProcessResult::Error;
            }

            segmentBuffer.setContinuationRecordType(m_parsers->tlsRecord->getRecordType());
            segmentBuffer.setContinuationRecordLength(recordPayloadLength);
            segmentBuffer.setAlertHeaderParsed(true);
            segmentBuffer.setContinuationRemainingLength(totalRecordLength - updatedBytesReceived);
        }

        const uint16_t remaining = segmentBuffer.getContinuationRemainingLength();
        if (remaining == 0)
        {
            break;
        }

        const uint16_t bytesToConsume = std::min(remaining, m_lengthLeftToParse);
        if (bytesToConsume == 0)
        {
            return RecordProcessResult::Segmented;
        }

        const uint16_t bytesReceived = segmentBuffer.getAlertRecordBytesReceived();
        if (bytesReceived < TLSRecordHeaderSize + 2)
        {
            const uint16_t bytesToStore = std::min<uint16_t>(bytesToConsume,
                                                              TLSRecordHeaderSize + 2 - bytesReceived);
            if (bytesToStore != 0)
            {
                std::memcpy(segmentBuffer.getBuffer().data() + bytesReceived,
                            m_parserLocationPtr,
                            bytesToStore);
                segmentBuffer.setBufferUsed(bytesReceived + bytesToStore);
            }
        }

        m_parserLocationPtr += bytesToConsume;
        m_lengthLeftToParse -= bytesToConsume;
        segmentBuffer.setAlertRecordBytesReceived(bytesReceived + bytesToConsume);
        segmentBuffer.setContinuationRemainingLength(remaining - bytesToConsume);
        segmentBuffer.setNextSequence(m_nextExpectedSeq);

        if (remaining > bytesToConsume)
        {
            return RecordProcessResult::Segmented;
        }
    }

    m_record = Record{};
    m_record.recordType = segmentBuffer.getContinuationRecordType();
    m_record.handshakeType = segmentBuffer.getContinuationHandshakeType();
    m_record.recordLength = segmentBuffer.getContinuationRecordLength();

    if (segmentBuffer.isAlertPlaintextVisible() &&
        segmentBuffer.getAlertAlignmentOffset() == 0 &&
        m_record.recordLength == 2 &&
        segmentBuffer.getBufferUsed() >= TLSRecordHeaderSize + 2)
    {
        uint16_t consumed = 0;
        const uint8_t *alertPayload = reinterpret_cast<const uint8_t *>(segmentBuffer.getBuffer().data()) + TLSRecordHeaderSize;
        if (m_parsers->alert->parse(alertPayload, 2, consumed) == ParsingStatus::Complete && consumed == 2)
        {
            m_record.alertLevel = m_parsers->alert->getAlertLevel();
            m_record.alertDescription = m_parsers->alert->getAlertDescription();
            m_record.readableAlert = true;
        }
    }

    const RecordProcessResult alignmentResult = updateAlertAlignment(
        segmentBuffer.isAlertPlaintextVisible(), m_record.recordLength);
    if (alignmentResult != RecordProcessResult::ContinueProcessing)
    {
        return alignmentResult;
    }

    if (m_records.size() >= maxRecords)
    {
        m_lastError = ParsingError::RecordsExceedDataViewerCapacity;
        finishSegmentBufferContinuation();
        return RecordProcessResult::StopProcessing;
    }
    m_records.push_back(m_record);
    finishSegmentBufferContinuation();
    return RecordProcessResult::ContinueProcessing;
}

Processor::RecordProcessResult Processor::processAlert(const TLSData &data)
{
    const bool plaintextVisible =
        data.hsState == TLSFlow::HsState::PreHandshake ||
        data.hsState == TLSFlow::HsState::InProgress;

    if (plaintextVisible &&
        m_record.recordLength == 2 &&
        m_pSegmentBuffer->getAlertAlignmentOffset() == 0 &&
        m_currentRecordPayloadLeft == 2 &&
        m_parsers->alert != nullptr)
    {
        uint16_t consumed = 0;
        if (m_parsers->alert->parse(m_currentRecordPayloadPtr, 2, consumed) == ParsingStatus::Complete &&
            consumed == 2)
        {
            m_record.alertLevel = m_parsers->alert->getAlertLevel();
            m_record.alertDescription = m_parsers->alert->getAlertDescription();
            m_record.readableAlert = true;
        }
    }

    return updateAlertAlignment(plaintextVisible, m_record.recordLength);
}

Processor::RecordProcessResult Processor::updateAlertAlignment(bool plaintextVisible, uint16_t recordLength)
{
    if (!plaintextVisible)
    {
        return RecordProcessResult::ContinueProcessing;
    }

    if ((recordLength & 1u) == 0)
    {
        if (m_pSegmentBuffer != &m_initialSegment &&
            m_pSegmentBuffer->getAlertAlignmentOffset() != 0)
        {
            m_pSegmentBuffer->setNextSequence(m_nextExpectedSeq);
        }
        return RecordProcessResult::ContinueProcessing;
    }

    if (m_pSegmentBuffer == &m_initialSegment)
    {
        bool allocated = false;
        if (!prepareSegmentBufferForContinuation(SegmentBuffer::ContinuationMode::None, allocated))
        {
            m_lastError = ParsingError::SegmentBufferAllocationFailure;
            return RecordProcessResult::StopProcessing;
        }
    }

    const uint8_t alignmentOffset = static_cast<uint8_t>(m_pSegmentBuffer->getAlertAlignmentOffset() ^ 1u);
    m_pSegmentBuffer->setAlertAlignmentOffset(alignmentOffset);
    if (alignmentOffset == 0)
    {
        deallocateSegmentBuff();
    }
    else
    {
        m_pSegmentBuffer->setNextSequence(m_nextExpectedSeq);
    }
    return RecordProcessResult::ContinueProcessing;
}

Processor::RecordProcessResult Processor::processNewRecord(const TLSData &data)
{
    bool unsegmentedClientHelloCompleted = false;

    if (data.hsState == TLSFlow::HsState::Unsupported)
    {
        return RecordProcessResult::EndOfPacket;
    }

    if (m_lengthLeftToParse < TLSRecordHeaderSize &&
        m_lengthLeftToParse != 0 &&
        m_parserLocationPtr != nullptr &&
        *m_parserLocationPtr == static_cast<uint8_t>(RecordTypesE::alertContent))
    {
        if (!startSegmentedAlert(data))
        {
            m_lastError = ParsingError::SegmentBufferAllocationFailure;
            return RecordProcessResult::StopProcessing;
        }
        return RecordProcessResult::Segmented;
    }

    // Parse TLS record header (uses m_parserLocationPtr, m_lengthLeftToParse but doesn't modify them)
    auto innerProcessResult = processTlsRecord();
    if (RecordProcessResult::ContinueProcessing != innerProcessResult)
    {
        return innerProcessResult;
    }

    // Get record type and length (use raw uint8_t to capture all record types including unknown)
    uint8_t recordType = m_parsers->tlsRecord->getRecordType();
    uint16_t recordPayloadLen = m_parsers->tlsRecord->getTlsRecordLength();
    uint16_t totalRecordLen = TLSRecordHeaderSize + recordPayloadLen;
    m_currentRecordIsSegmented = m_lengthLeftToParse < totalRecordLen;

    // Initialize record-level tracking
    m_currentRecordPayloadPtr = m_parserLocationPtr + TLSRecordHeaderSize;
    const uint16_t availablePayloadLength = static_cast<uint16_t>(m_lengthLeftToParse - TLSRecordHeaderSize);
    m_currentRecordPayloadLeft = std::min(recordPayloadLen, availablePayloadLength);

    m_record = Record{};
    m_record.recordType = recordType;
    m_record.recordLength = recordPayloadLen;

    if (recordType == static_cast<uint8_t>(RecordTypesE::alertContent))
    {
        if (m_currentRecordIsSegmented)
        {
            if (!startSegmentedAlert(data, recordPayloadLen))
            {
                m_lastError = ParsingError::SegmentBufferAllocationFailure;
                return RecordProcessResult::StopProcessing;
            }
            return RecordProcessResult::Segmented;
        }

        innerProcessResult = processAlert(data);
        if (RecordProcessResult::ContinueProcessing != innerProcessResult)
        {
            return innerProcessResult;
        }
    }

    const bool opaqueRecord =
        (data.hsState == TLSFlow::HsState::PreHandshake &&
         recordType == static_cast<uint8_t>(RecordTypesE::applicationDataContent)) ||
        (data.hsState == TLSFlow::HsState::InProgress &&
         ((TLSFlow::isTLS13(data.tlsVersionMask) &&
           recordType == static_cast<uint8_t>(RecordTypesE::applicationDataContent)) ||
          recordType == static_cast<uint8_t>(RecordTypesE::changeCipherSpecContent))) ||
        (data.hsState == TLSFlow::HsState::ChangeCipherSpec &&
         recordType == static_cast<uint8_t>(RecordTypesE::handshakeContent));

    if (m_currentRecordIsSegmented && opaqueRecord)
    {
        bool allocated = false;
        if (!prepareSegmentBufferForContinuation(SegmentBuffer::ContinuationMode::OpaqueRecordTracking, allocated))
        {
            m_lastError = ParsingError::SegmentBufferAllocationFailure;
            return RecordProcessResult::StopProcessing;
        }
        m_pSegmentBuffer->setNextSequence(m_nextExpectedSeq);
        m_pSegmentBuffer->setContinuationRemainingLength(totalRecordLen - m_lengthLeftToParse);
        m_pSegmentBuffer->setContinuationRecordType(recordType);
        m_pSegmentBuffer->setContinuationHandshakeType(m_record.handshakeType);
        m_pSegmentBuffer->setContinuationRecordLength(recordPayloadLen);
        m_diagnosticsCounters->incEvent(DiagnosticsEvents::OpaqueRecordContinuationStarted);
        return RecordProcessResult::Segmented;
    }
    
    const bool shouldParseHandshake =
        data.hsState == TLSFlow::HsState::PreHandshake ||
        data.hsState == TLSFlow::HsState::InProgress;
    if (shouldParseHandshake &&
        recordType == static_cast<uint8_t>(RecordTypesE::handshakeContent))
    {
        // Parse handshake header
        innerProcessResult = processHandshake();
        if (RecordProcessResult::ContinueProcessing != innerProcessResult)
        {
            return innerProcessResult;
        }

        // Get handshake type
        uint8_t hsType = m_parsers->handshake->getHandshakeType();
        m_record.handshakeType = hsType;

        if (hsType != static_cast<uint8_t>(HandShakeTypesE::client_hello))
        {
            // Non-ClientHello handshake - log and continue to record it
            DIAG_LOGGER_DEBUG_MODE_INJECT(PROCESSOR_TLS_RECORD_NON_CLIENT_HELLO, hsType, m_currentRecordPayloadLeft);
        }
        else // ClientHello processing
        {
            
            if (!m_parsers->tlsRecord->isSupportedVersion())
            {
                DIAG_LOGGER_DEBUG_MODE_INJECT(PROCESSOR_TLS_RECORD_UNSUPPORTED_VER, m_parsers->tlsRecord->getTlsVersion(), 0, m_lengthLeftToParse);
                m_diagnosticsCounters->incEvent(DiagnosticsEvents::UnsupportedVersion);
                m_lastError = ParsingError::UnsupportedTlsVersion;
                return RecordProcessResult::Error;
            }

            const uint32_t handshakeRecordLength =
                m_parsers->handshake->getHandshakeLength() + TLSHandshakeHeaderSize;
            const uint16_t recordLength = m_parsers->tlsRecord->getTlsRecordLength();
            if (handshakeRecordLength < recordLength)
            {
                DIAG_LOGGER_INJECT(PROCESSOR_PARSE_FAILED, 
                    "Handshake length mismatch: handshake=%u, record=%u", 
                    handshakeRecordLength,
                    recordLength);

                m_lastError = ParsingError::HandshakeLengthNotMatchingRecordLength;
                return RecordProcessResult::StopProcessing;
            }

            // Multiple ClientHello is anomalous - reject
            if (m_clientHelloParsingStatus == IClientHelloDataViewer::ClientHelloParsingStatus::Complete)
            {
                DIAG_LOGGER_DEBUG_MODE_INJECT(PROCESSOR_PARSE_FAILED, "Multiple ClientHello detected", 0, m_currentRecordPayloadLeft);
                m_lastError = ParsingError::MultipleClientHello;
                return RecordProcessResult::Error;
            }
            
            // ClientHello detected
            m_clientHelloParsingStatus = IClientHelloDataViewer::ClientHelloParsingStatus::Partial;
            
            // Initialize digest (version will be hashed after parsing ClientHello)
            m_pDigest->init();

            // Process ClientHello body
            const bool recordLevelFragmented = handshakeRecordLength > recordLength &&
                                               !m_currentRecordIsSegmented;
            innerProcessResult = processClientHello(recordLevelFragmented);
            if (RecordProcessResult::ContinueProcessing != innerProcessResult)
            {
                // Error or Segmented - return without advancing (segmentation saves state)
                return innerProcessResult;
            }

            if (handshakeRecordLength > recordLength)
            {
                if (recordLevelFragmented)
                {
                    m_diagnosticsCounters->incEvent(DiagnosticsEvents::UnsupportedRecordLevelFragmentation);
                    return RecordProcessResult::Unsupported;
                }

                m_lastError = ParsingError::HandshakeLengthNotMatchingRecordLength;
                return RecordProcessResult::StopProcessing;
            }

            // Hash ClientHello version (now available from ClientHelloParser)
            DIAG_LOGGER_INJECT_DEBUG_ONLY(PROCESSOR_CALC_ADD, "ClientHello version");
            if (!appendOpenParentheses())
            {
                return RecordProcessResult::StopProcessing;
            }
            if (!calcHash(m_parsers->clientHello->getVersion()))
            {
                DIAG_LOGGER_DEBUG_MODE_INJECT(PROCESSOR_HS_UNSUPPORTED_VER, m_parsers->clientHello->getVersion());
                m_diagnosticsCounters->incEvent(DiagnosticsEvents::UnsupportedVersion);
                return RecordProcessResult::StopProcessing;
            }
            if (!appendClosingParentheses())
            {
                return RecordProcessResult::StopProcessing;
            }

            /* Save the version from the ClientHello, may be overridden later if supported_versions extension exists */
            lastVersion = m_parsers->clientHello->getVersion();

            m_numOfCompressions = m_parsers->clientHello->getCompressionLen();

            innerProcessResult = processCiphers();
            if (RecordProcessResult::ContinueProcessing != innerProcessResult)
            {
                if (innerProcessResult == RecordProcessResult::Error)
                {
                    // m_lastError already set by processCiphers
                    return RecordProcessResult::Error;
                }
                // Non-error stop - will record below
            }
            else
            {
                m_numOfCipherSuites = m_parsers->cipherSuites->getNumOfSuites();

                innerProcessResult = processExtensions();
                if (RecordProcessResult::ContinueProcessing != innerProcessResult)
                {
                    if (RecordProcessResult::Segmented != innerProcessResult)
                    {
                        if (RecordProcessResult::Unsupported != innerProcessResult)
                        {
                            m_diagnosticsCounters->incEvent(DiagnosticsEvents::ExtensionParserFailures);
                        }
                        // m_lastError already set by processExtensions
                        return innerProcessResult;
                    }
                    
                    if (m_state == ProcessorState::FirstSegment)
                    {
                        // Extensions are segmented - save state for continuation
                        m_pSegmentBuffer->setNumOfCompressions(m_numOfCompressions);
                        m_pSegmentBuffer->setLastVersion(lastVersion);
                        m_pSegmentBuffer->setNumOfCiphers(m_numOfCipherSuites);
                        m_pSegmentBuffer->setExtensionRemainingLength(
                            m_parsers->clientHello->getExtensionsLen() - m_pSegmentBuffer->getBufferUsed());
                        m_diagnosticsCounters->incEvent(DiagnosticsEvents::FirstSegmentProcessSuccess);
                        return RecordProcessResult::Segmented;
                    }
                    // Non-error, non-segmented stop - will record below
                }
                else
                {
                    if (!m_supportedVersionsParsed)
                    {
                        m_supportedVersionsBitMask = tlsVersionMaskFromWireVersion(m_parsers->clientHello->getVersion());
                    }
                    m_pDigest->finalize(m_fp);
                    m_clientHelloParsingStatus = IClientHelloDataViewer::ClientHelloParsingStatus::Complete;
                    unsegmentedClientHelloCompleted = true;
                }
            }
        }
    }

    // Add record to the list
    if(m_records.size() >= maxRecords){
        m_lastError = ParsingError::RecordsExceedDataViewerCapacity;
        return RecordProcessResult::StopProcessing;
    }
    m_records.push_back(m_record);

    if (unsegmentedClientHelloCompleted)
    {
        m_diagnosticsCounters->incEvent(DiagnosticsEvents::UnsegmentedClientHelloCompleted);
    }
    

    if(m_lengthLeftToParse < totalRecordLen)
    {
        // Incomplete record 
        m_diagnosticsCounters->incEvent(DiagnosticsEvents::FirstSegmentProcessSuccess);
        return RecordProcessResult::Segmented;
    }

    // Advance segment-level pointers past this entire record (single advancement point)
    m_parserLocationPtr += totalRecordLen;
    m_lengthLeftToParse -= totalRecordLen;

    return innerProcessResult;
}

Processor::RecordProcessResult Processor::processExtensions()
{
    const uint8_t* extensionsPtr = nullptr;
    uint16_t extensionsLen = 0;
    
    switch (getState())
    {
        /* Normal single segment processing */
        case ProcessorState::SingleSegment:
            extensionsLen = m_parsers->clientHello->getExtensionsLen();
            extensionsPtr = m_parsers->clientHello->getpExtensions();
            m_pExtensionLocators->reset();
            break;
            
        /* This is the first segment treated, work with first partial part */
        case ProcessorState::FirstSegment:
        {
            /* Reset the extensions array */
            m_pExtensionLocators->reset();

            /* Validate total extension length fits in segment buffer */
            extensionsLen = m_parsers->clientHello->getExtensionsLen();
            if (extensionsLen > SegmentBufferSize)
            {
                m_diagnosticsCounters->incEvent(DiagnosticsEvents::UnsupportedExtensionsExceedBuffer);
                deallocateSegmentBuff();
                return RecordProcessResult::Unsupported;
            }

            uint16_t nextLength = m_pSegmentBuffer->getContinuationRemainingLength();
            if (extensionsLen < nextLength) {
                // Would underflow - invalid state
                m_diagnosticsCounters->incEvent(DiagnosticsEvents::SegmentedStateSecondSegmentTooShort);
                m_lastError = ParsingError::SegmentMismatch;
                deallocateSegmentBuff();
                return RecordProcessResult::StopProcessing;
            }            
            
            /* Calculate length to copy from the partial extensions */
            uint16_t toCopy = extensionsLen - nextLength;
            if(toCopy > m_lengthLeftToParse)
            {
                m_diagnosticsCounters->incEvent(DiagnosticsEvents::SegmentedStateSecondSegmentTooShort);
                m_lastError = ParsingError::SegmentMismatch;
                deallocateSegmentBuff();
                return RecordProcessResult::StopProcessing;
            }
            /* Copy the partial extensions to the segment buffer for further processing */
            std::memcpy(m_pSegmentBuffer->getBuffer().data(), m_parsers->clientHello->getpExtensions(), toCopy);
            m_pSegmentBuffer->setBufferUsed(toCopy);

            return RecordProcessResult::Segmented;
        }

        /* This is the last segment treated, work with the saved segment buffer */
        case ProcessorState::LastSegment:
        {
            /* Copy only the bytes needed for ClientHello completion to the segment buffer */
            uint16_t bytesNeeded = m_pSegmentBuffer->getNextLength();
            std::memcpy(m_pSegmentBuffer->getBuffer().data() + m_pSegmentBuffer->getBufferUsed(), m_parserLocationPtr, bytesNeeded);
            m_pSegmentBuffer->setBufferUsed(m_pSegmentBuffer->getBufferUsed() + bytesNeeded);

            /* Set local pointers to parse from the segment buffer */
            extensionsPtr = reinterpret_cast<const uint8_t*>(m_pSegmentBuffer->getBuffer().data());
            extensionsLen = m_pSegmentBuffer->getBufferUsed();
            break;
        }
        default:
            m_lastError = ParsingError::ExtensionParsingFailure;
            return RecordProcessResult::Error;
    }

    /* Parse extensions */
    auto res = parseAllExtensions(extensionsPtr, extensionsLen);
    if (res != RecordProcessResult::ContinueProcessing)
        return res;

    // Sort the extensions according to type, then length, then value, greased last
    m_pExtensionLocators->sort();
    m_numOfExtensions = m_pExtensionLocators->getNumOfExtensions();
    return feedExtensionsFP();
}

Processor::RecordProcessResult Processor::processSni()
{
    /* Parse SNI extension */
    uint16_t consumedSni;
    if (m_parsers->sni->parse(m_sniExtension.pValue, m_sniExtension.length, consumedSni) != ParsingStatus::Complete)
    {
        DIAG_LOGGER_DEBUG_MODE_INJECT(PROCESSOR_PARSE_FAILED, "SNI", consumedSni, m_sniExtension.length);
        m_lastError = ParsingError::ExtensionParsingFailure;
        return RecordProcessResult::Error;
    }

    /* Validate number of SNIs */
    uint16_t numOfSnis = m_parsers->sni->getNumOfSni();
    if (numOfSnis > 2 || numOfSnis == 0)
    {
        m_lastError = ParsingError::MalformedSniExtension;
        return RecordProcessResult::Error;
    }

    /* Validate SNI type */
    uint16_t sniType = m_parsers->sni->getSniType(0);
    if (static_cast<SniTypeE>(sniType) != SniTypeE::hostName)
    {
        DIAG_LOGGER_DEBUG_MODE_INJECT(PROCESSOR_NON_HOST_SNI, sniType);
        return RecordProcessResult::ContinueProcessing;
    }

    /* Validate SNI length */
    uint16_t sniLen = m_parsers->sni->getSniLen(0);
    if (sniLen == 0 || sniLen > max_sni_length)
    {
        DIAG_LOGGER_DEBUG_MODE_INJECT(PROCESSOR_SNI_TOO_LONG, sniLen);
        m_lastError = ParsingError::MalformedSniExtension;
        return RecordProcessResult::Error;
    }

    /* Size is consistent */
    auto sni = reinterpret_cast<const char*>(m_parsers->sni->getpSni(0));
    if (sniLen != strnlen(sni, sniLen))
    {
        DIAG_LOGGER_DEBUG_MODE_INJECT(PROCESSOR_SNI_INCONSISTENT);
        m_lastError = ParsingError::MalformedSniExtension;
        return RecordProcessResult::Error;
    }

    /* Save the SNI */
    m_sni.set(reinterpret_cast<const char*>(m_parsers->sni->getpSni(0)), sniLen);
    m_bSniExtensionParsed = true;
    return RecordProcessResult::ContinueProcessing;
}

// Go over all sorted extensions and inject the relevant parts into the FP hash
Processor::RecordProcessResult Processor::feedExtensionsFP()
{
    // Add a square parentheses around the extensions part, also in case no extensions exist
    if (!appendOpenSqParentheses()) {
        return RecordProcessResult::StopProcessing;
    }
    const uint8_t *pVal;
    for (uint16_t e = 0; e < m_pExtensionLocators->getNumOfExtensions(); e++)
    {
        const ExtensionLocator& extensionLocator = m_pExtensionLocators->get(e);

        // Add extension type to fp hash
        if (!appendOpenParentheses())
        {
            return RecordProcessResult::StopProcessing;
        }
        DIAG_LOGGER_INJECT_DEBUG_ONLY(PROCESSOR_CALC_ADD, "ExtensionTypeE");

        if (!calcHash(static_cast<uint16_t>(extensionLocator.type)))
        {
            return RecordProcessResult::StopProcessing;
        }

        if (isExtensionContentIncludedInFp(extensionLocator.type))
        {
            // Add extension Length to fp hash
            DIAG_LOGGER_INJECT_DEBUG_ONLY(PROCESSOR_CALC_ADD, "extensionLen");
            if (!calcHash(static_cast<uint16_t>(extensionLocator.length)))
            {
                return RecordProcessResult::StopProcessing;
            }
            DIAG_LOGGER_INJECT_DEBUG_ONLY(PROCESSOR_EXTENSION_REQUIORED, "Extension required");
            switch (extensionLocator.type)
            {
                case ExtensionTypeE::supported_groups:
                { // supported groups
                    auto processRes = feedSuppGroupsFP(extensionLocator);
                    if (processRes != RecordProcessResult::ContinueProcessing)
                    {
                        return processRes;
                    }
                    break;
                }
                case ExtensionTypeE::supported_versions:
                { // supported versions
                    auto processRes = feedSuppVersionsFP(extensionLocator);
                    if (processRes != RecordProcessResult::ContinueProcessing)
                    {
                        return processRes;
                    }
                    break;
                }
                default: DIAG_LOGGER_INJECT_DEBUG_ONLY(PROCESSOR_CALC_ADD, "General extension");
                    if (!calcHash(extensionLocator.pValue, extensionLocator.length))
                    {
                        return RecordProcessResult::StopProcessing;
                    }
            }
        }

        if (!appendClosingParentheses())
        {
            return RecordProcessResult::StopProcessing;
        }
    }
    // Close the square parentheses
    if (!appendClosingSqParentheses())
    {
        return RecordProcessResult::StopProcessing;
    }
    return RecordProcessResult::ContinueProcessing;
}

Processor::RecordProcessResult Processor::feedSuppVersionsFP(const ExtensionLocator& extensionLocator)
{
    uint16_t consumedVersions;
    if (ParsingStatus::Complete != m_parsers->supportedVersions->parse(extensionLocator.pValue, extensionLocator.length, consumedVersions))
    {
        DIAG_LOGGER_DEBUG_MODE_INJECT(PROCESSOR_PARSE_FAILED, "Supported Versions", consumedVersions, extensionLocator.length);
        m_lastError = ParsingError::ExtensionParsingFailure;
        return RecordProcessResult::Error;
    }
    DIAG_LOGGER_INJECT_DEBUG_ONLY(PROCESSOR_CALC_ADD, "Length of Supported Versions");
    if (!calcHash(m_parsers->supportedVersions->getLength()))
    {
        return RecordProcessResult::StopProcessing;
    }

    uint16_t supportedVersions = m_parsers->supportedVersions->getNumOfVersions(), i = 0;
    bool versionFound = false;
    m_supportedVersionsParsed = true;
    for (; supportedVersions--; i++)
    {
        // Skip GREASE values when setting the lastVersion
        auto version = m_parsers->supportedVersions->getVersion(i);
        if(!versionFound && !isGreaseValue(version))
        {
            lastVersion = version;
            versionFound = true;
        }
        DIAG_LOGGER_INJECT_DEBUG_ONLY(PROCESSOR_CALC_ADD, "Supported Version");
        // Add each supported version to the hash
        m_supportedVersionsBitMask |= tlsVersionMaskFromWireVersion(version);
        if (!calcHash(deGrease(version)))
        {
            return RecordProcessResult::StopProcessing;
        }
    }

    return RecordProcessResult::ContinueProcessing;
}

Processor::RecordProcessResult Processor::feedSuppGroupsFP(const ExtensionLocator& extensionLocator)
{
    uint16_t consumedGroup;

    if (ParsingStatus::Complete != m_parsers->supportedGroups->parse(extensionLocator.pValue, extensionLocator.length, consumedGroup))
    {
        DIAG_LOGGER_DEBUG_MODE_INJECT(PROCESSOR_PARSE_FAILED, "Supported Groups", consumedGroup, m_lengthLeftToParse);
        m_lastError = ParsingError::ExtensionParsingFailure;
        return RecordProcessResult::Error;
    }
    DIAG_LOGGER_INJECT_DEBUG_ONLY(PROCESSOR_CALC_ADD, "Length of Supported Groups");
    if (!calcHash(m_parsers->supportedGroups->getGroupsLen()))
    {
        return RecordProcessResult::StopProcessing;
    }

    uint16_t supportedGroups = m_parsers->supportedGroups->getNumOfGroups(), i = 0;
    for (; supportedGroups--; i++)
    {
        DIAG_LOGGER_INJECT_DEBUG_ONLY(PROCESSOR_CALC_ADD, "Supported Group");
        if (!calcHash(deGrease(m_parsers->supportedGroups->getGroup(i))))
        {
            return RecordProcessResult::StopProcessing;
        }
    }
    return RecordProcessResult::ContinueProcessing;
}

Processor::RecordProcessResult Processor::parseAllExtensions(const uint8_t* extensionsPtr, uint16_t extensionsLenLeftToParse)
{
    uint16_t consumed;
    uint16_t offset = 0;
    
    // Parse and fill the extensions for sorting them before adding to the hash calculation
    while (extensionsLenLeftToParse > 0)
    {
        /* Parse next extension */
        auto parsingRes = m_parsers->extensions->parse(extensionsPtr + offset, extensionsLenLeftToParse, consumed);
        if (parsingRes != IExtensionsParser::Status::Complete)
        {
            DIAG_LOGGER_DEBUG_MODE_INJECT(PROCESSOR_PARSE_FAILED, "Extensions incomplete result inconsistency", getState(), extensionsLenLeftToParse);
            if(parsingRes == IExtensionsParser::Status::Failure)
            {
                m_lastError = ParsingError::ExtensionParsingFailure;
                return RecordProcessResult::Error;
            }
            m_lastError = ParsingError::ExtensionParsingFailure;
            return RecordProcessResult::Error;
        }

        /* Logic check; consumed bytes must consist with parsed extension size */
        if (m_parsers->extensions->getSingleExtensionTotalLength() != consumed)
        {
            DIAG_LOGGER_DEBUG_MODE_INJECT(PROCESSOR_EXTENSIONS_LENGTH_INVALID, m_parsers->extensions->getSingleExtensionTotalLength(), consumed, extensionsLenLeftToParse);
            m_lastError = ParsingError::ExtensionParsingFailure;
            return RecordProcessResult::Error;
        }

        /* Add the parsed extension to the extensions array for later sorting */
        if (!m_pExtensionLocators->add({deGrease(static_cast<uint16_t>(m_parsers->extensions->getExtensionType())), m_parsers->extensions->getExtensionDataLength(), m_parsers->extensions->getExtensionValuePtr()}))
        {
            DIAG_LOGGER_DEBUG_MODE_INJECT(PROCESSOR_PARSE_FAILED, "Extensions array insert failed", consumed, extensionsLenLeftToParse);
            return RecordProcessResult::StopProcessing;
        }

        if (m_pExtensionLocators->getLast().getType() == ExtensionTypeE::early_data)
        {
            m_earlyDataOffered = true;
        }

        /* Check for SNI extension and save its locator for later direct access */
        if (m_pExtensionLocators->getLast().getType() == ExtensionTypeE::server_name)
        {
            m_sniExtension = m_pExtensionLocators->getLast();
            if(processSni() == RecordProcessResult::Error)
            {
                // m_lastError already set by processSni
                return RecordProcessResult::Error;
            }
        }

        /* Move to next extension */
        extensionsLenLeftToParse -= consumed;
        offset += consumed;
    }

    return RecordProcessResult::ContinueProcessing;
}

Processor::RecordProcessResult Processor::processCiphers()
{
    uint16_t consumed;
    const uint8_t *pCipherSuites = m_parsers->clientHello->getpCipherSuites();
    uint16_t cipherSuitesLen = m_parsers->clientHello->getCipherSuitesLen();

    if (cipherSuitesLen)
    {
        if (ParsingStatus::Complete != m_parsers->cipherSuites->parse(pCipherSuites, cipherSuitesLen, consumed))
        {
            DIAG_LOGGER_DEBUG_MODE_INJECT(PROCESSOR_PARSE_FAILED, "Cipher Suites", consumed, cipherSuitesLen);
            m_lastError = ParsingError::CipherParsingFailure;
            return RecordProcessResult::Error;
        }
        if (!appendOpenParentheses())
        {
            return RecordProcessResult::StopProcessing;
        }
        for (int i = 0; cipherSuitesLen -= 2; i++)
        {
            DIAG_LOGGER_INJECT_DEBUG_ONLY(PROCESSOR_CALC_ADD, "CipherSuite");
            if (!calcHash(deGrease(m_parsers->cipherSuites->getSuite(i))))
            {
                return RecordProcessResult::StopProcessing;
            }
        }
        if (!appendClosingParentheses())
        {
            return RecordProcessResult::StopProcessing;
        }
    }
    return RecordProcessResult::ContinueProcessing;
}

Processor::RecordProcessResult Processor::processClientHello(bool recordLevelFragmented)
{
    uint16_t consumed;
    uint32_t expectedLength = m_parsers->handshake->getHandshakeLength();
    auto status = m_parsers->clientHello->parse(m_currentRecordPayloadPtr, m_currentRecordPayloadLeft, expectedLength, consumed);

    if (IClientHelloParser::Status::Complete != status)
    {
        if (recordLevelFragmented &&
            (IClientHelloParser::Status::IncompleteClientHelloPrefix == status ||
             IClientHelloParser::Status::IncompleteClientHelloBody == status ||
             IClientHelloParser::Status::IncompleteExtensionSection == status))
        {
            m_diagnosticsCounters->incEvent(DiagnosticsEvents::UnsupportedRecordLevelFragmentation);
            return RecordProcessResult::Unsupported;
        }

        /* TLS is segmented */
        if (IClientHelloParser::Status::IncompleteExtensionSection == status)
        {
            bool allocated = false;
            if (!prepareSegmentBufferForContinuation(SegmentBuffer::ContinuationMode::ClientHelloExtensionReassembly,
                                                      allocated))
            {
                DIAG_LOGGER_DEBUG_MODE_INJECT(PROCESSOR_ERROR_ALLOCATE_SEGSTATE);
                m_diagnosticsCounters->incEvent(DiagnosticsEvents::SegmentedStateEntriesAllocationFailures);
                m_lastError = ParsingError::SegmentBufferAllocationFailure;
                return RecordProcessResult::StopProcessing;
            }

            /* Initialize segment buffer */
            setState(ProcessorState::FirstSegment);
            m_pSegmentBuffer->setNextSequence(m_nextExpectedSeq);
            m_pSegmentBuffer->setNextLength(m_parsers->tlsRecord->getTlsRecordLength() + TLSRecordHeaderSize - m_packetPayloadLen);
            m_pSegmentBuffer->setStepBufferedLength(m_packetPayloadLen);
            m_pSegmentBuffer->setTotalRecordLength(m_parsers->tlsRecord->getTlsRecordLength());
            m_pSegmentBuffer->setContinuationMode(SegmentBuffer::ContinuationMode::ClientHelloExtensionReassembly);
            m_pSegmentBuffer->setClientHelloSegmentCount(1);
            m_pSegmentBuffer->setContinuationRemainingLength(
                m_parsers->tlsRecord->getTlsRecordLength() + TLSRecordHeaderSize - m_lengthLeftToParse);
            m_pSegmentBuffer->setContinuationRecordType(static_cast<uint8_t>(RecordTypesE::handshakeContent));
            m_pSegmentBuffer->setContinuationHandshakeType(static_cast<uint8_t>(HandShakeTypesE::client_hello));
            m_pSegmentBuffer->setContinuationRecordLength(m_parsers->tlsRecord->getTlsRecordLength());
            m_pSegmentBuffer->copyDigest(*m_pDigest); // copy
            m_pDigest = &(m_pSegmentBuffer->getDigest());
            m_pExtensionLocators = m_pSegmentBuffer->getLocatorsArray();
            DIAG_LOGGER_DEBUG_MODE_INJECT(PROCESSOR_ALLOCATED_SEGMENT, m_pSegmentBuffer->getNextLength(), m_pSegmentBuffer->getNextSequence());
            m_diagnosticsCounters->incEvent(DiagnosticsEvents::ClientHelloContinuationStarted);
        }
        else if ((IClientHelloParser::Status::IncompleteClientHelloPrefix == status ||
              IClientHelloParser::Status::IncompleteClientHelloBody == status) &&
                 m_currentRecordIsSegmented)
        {
            m_diagnosticsCounters->incEvent(DiagnosticsEvents::UnsupportedEarlyClientHelloBody);
            return RecordProcessResult::Unsupported;
        }
        else
        {
            DIAG_LOGGER_DEBUG_MODE_INJECT(PROCESSOR_PARSE_FAILED, "Client-Hello", consumed, m_currentRecordPayloadLeft);
            m_lastError = ParsingError::ClientHelloBodyInvalid;
            return RecordProcessResult::Error;
        }
    }
    
    // Advance record-level pointers
    m_currentRecordPayloadPtr += consumed;
    m_currentRecordPayloadLeft -= consumed;

    return RecordProcessResult::ContinueProcessing;
}

Processor::RecordProcessResult Processor::processHandshake()
{
    uint16_t consumed;
    if (ParsingStatus::Complete != m_parsers->handshake->parse(m_currentRecordPayloadPtr, m_currentRecordPayloadLeft, consumed))
    {
        DIAG_LOGGER_DEBUG_MODE_INJECT(PROCESSOR_PARSE_FAILED, "Handshake", consumed, m_currentRecordPayloadLeft);
        m_lastError = ParsingError::MalformedHandshake;
        return RecordProcessResult::Error;
    }
    // Advance record-level pointers
    m_currentRecordPayloadPtr += consumed;
    m_currentRecordPayloadLeft -= consumed;
    return RecordProcessResult::ContinueProcessing;
}

Processor::RecordProcessResult Processor::processTlsRecord()
{
    uint16_t consumed;
    if (ParsingStatus::Complete != m_parsers->tlsRecord->parse(m_parserLocationPtr, m_lengthLeftToParse, consumed))
    {
        DIAG_LOGGER_DEBUG_MODE_INJECT(PROCESSOR_PARSE_FAILED, "TLS record", consumed, m_lengthLeftToParse);
        m_lastError = ParsingError::MalformedTlsRecord;
        return RecordProcessResult::Error;
    }

    // Validate record payload length against RFC 5246 Section 6.2.1 maximum (2^14 bytes)
    uint16_t recordPayloadLen = m_parsers->tlsRecord->getTlsRecordLength();
    if (recordPayloadLen > MaxTlsRecordPayloadSize) {
        m_lastError = ParsingError::RecordLengthExceedsData;
        return RecordProcessResult::Error;
    }

    if(!m_parsers->tlsRecord->isSupportedRecordType())
    {
        m_lastError = ParsingError::UnsupportedRecordType;
        return RecordProcessResult::StopProcessing;
    }

    /* Save the TLS version from the record layer only if ClientHello not yet processed.
       Once ClientHello is parsed, lastVersion comes from handshake/supported_versions extension. */
    if (m_clientHelloParsingStatus == IClientHelloDataViewer::ClientHelloParsingStatus::None)
    {
        lastVersion = m_parsers->tlsRecord->getTlsVersion();
    }

    // Note: We don't advance segment pointers here - processSingleRecord manages that
    return RecordProcessResult::ContinueProcessing;
}

using namespace TLSFlow::NSProcessor::Engine;
using namespace TLSFlow::NSDiagnostics::Common;

/// --------------------------
/// Concrete
/// --------------------------
SegmentBufferManager::SegmentBufferManager(
	SessionTable *sessionTable,
	NSDiagnostics::Engine::IDiagnosticsCounters& diagnosticsCounters):
	sessionTable{sessionTable}, diagnosticsCounters{diagnosticsCounters} {}

/// --------------------------
/// ISegmentBufferManager
/// --------------------------
bool SegmentBufferManager::allocate(const SessionId &id, Handler &&handler)
{
	/* Already exist? */
	auto mapItr = map.find(id);
	if(mapItr != map.end())
		return false;

	/* Fifo table is full? */
	if(map.isFull() == true)
	{
		diagnosticsCounters.incEvent(DiagnosticsEvents::SegmentBufferFull);

		/* Get oldest segment buffer */
		mapItr = map.begin();

		/* Fetch associated session of oldest segment buffer */
		auto session = sessionTable->getSessionByIndex(mapItr->first);
		if(session == nullptr)
			return false;

		/* Remove observer from session */
		session->deleteObserver(this, Events::STM_EVENT_AGING);

		/* Release oldest segment buffer */
		map.pop();
	}

	/* Insert into fifo map */
	auto res = map.insert({id, {}});
	if(res.second == false)
		return false;

	/* Fetch the new session */
	auto session = sessionTable->getSessionByIndex(id);
	if(session == nullptr)
		return false;

	/* Register observer to the new session */
	if(session->registerObserver(this, Events::STM_EVENT_AGING, AgingPriorityTLSSegmentBuffer) != 0)
	{
		map.erase(id);
		return false;
	}

	/* Invoke client handler */
	handler(res.first->second);
	return true;
}

bool SegmentBufferManager::deallocate(const SessionId &id)
{
	/* Doesn't exist? */
	auto itr = map.find(id);
	if(itr == map.end())
		return false;

	/* Release */
	if(map.erase(id) == 0)
		return false;

	/* Unregister observer */
	auto session = sessionTable->getSessionByIndex(id);
	if(session == nullptr)
		return false;

	session->deleteObserver(this, Events::STM_EVENT_AGING, AgingPriorityTLSSegmentBuffer);

	return true;
}

bool SegmentBufferManager::access(const SessionId &id, Handler &&handler)
{
	/* Doesn't exist? */
	auto itr = map.find(id);
	if(itr == map.end())
		return false;

	/* Invoke client handler */
	handler(itr->second);
	return true;
}

/// --------------------------
/// Observer
/// --------------------------
int SegmentBufferManager::onEvent(Session *session, DPFrame *packet, int event)
{
	/* Validate event is aging */
	if(event != Events::STM_EVENT_AGING)
		return ObserverResult::RET_OBSERVER_CONTINUE;

	/* Release */
	map.erase(session->getSessionIndex());
	return ObserverResult::RET_OBSERVER_CONTINUE;
}

                    return (m_offset + inc == 0) || isOffsetInRange(m_offset + inc - 1);
                }

                inline bool canIncOffset(uint16_t inc) const {
                    uint32_t sum = (uint32_t)m_offset + (uint32_t)inc;
                    if (sum > UINT16_MAX) return false;
                    return isOffsetInRange((uint16_t)sum);
                }

                inline bool incOffset(uint16_t inc) {
                    if (!canIncOffset(inc)) return false;
                    m_offset += inc;
                    return true;
                }

                inline bool setOffset(uint16_t offset) {
                    if (!isOffsetInRange(offset)) return false;
                    m_offset = offset;
                    return true;
                }

IClientHelloParser::Status ClientHelloParser::parse(const uint8_t *const payload, uint16_t bufferLen, uint32_t expectedLength, uint16_t &consumed) {

    m_status = Status::Failure;

    if (!payload) {
        return m_status;
    }

    if (bufferLen < m_minimalLength) {
		m_status = Status::IncompleteClientHelloPrefix;
        consumed = bufferLen;
        return m_status;
    }

    if (!init(payload, bufferLen)){
        return m_status;
    }

    // Parse version (first field in ClientHello body)
    if (!(setOffset(versionOffset)) ){
        return m_status;
    }
    m_hello.version = getDataAtOffsetU16();

    // Move to Random field
    if (!incOffset(versionLength)) {
        return m_status;
    }
    // Save pointer to random
    m_hello.pRandom = getDataPtrAtOffset();

    // Move to Session ID length identifier
    if (!incOffset(randomLength)) {
        return m_status;
    }
    // Read Session ID length (guaranteed by minimalHelloLength=37)
    m_hello.sidLen  = getDataAtOffsetU8();

    if (!incOffset(sessionIdLenFiledLen)){
        return m_status;
    }
    // Save pointer to Session ID
    m_hello.pSid = getDataPtrAtOffset();

    // Check if Session ID data is within payload (variable length)
    if (!incOffset(m_hello.sidLen)){
        return m_status;
    }
    // Read cipher suites length field (guaranteed by minimalHelloLength=37)
    m_hello.suitesLen = getDataAtOffsetU16() + suitesLengthFieldLen;

    // Make sure that cipher suites length is even (each cipher suite is 2 bytes long)
    if (m_hello.suitesLen % 2) {
      return m_status;
    }

    // Save pointer to Cipher Suites (including ciphers length field)
    m_hello.pCipherSuits = getDataPtrAtOffset();

    // Check if all ciphers are in payload and skip
    if (!incOffset(m_hello.suitesLen)) {
        return m_status;
    }

    // Check if compression methods length filed is within payload
    if (!canReadLength(compressionsLengthFieldLen)){
        return m_status;
    }
    // Read compression methods length field
    m_hello.compressionMethodsLen = getDataAtOffsetU8();

    // Skip compression methods length field
    if (!incOffset(compressionsLengthFieldLen)){
        return m_status;
    }

    // Save pointer to Compression Methods
    m_hello.pCompressionMethods = getDataPtrAtOffset();

    m_hello.extensionLen = 0;
    m_hello.pExtensions  = nullptr;

    // Check if extensions exist by comparing against expectedLength
    // (no extensions means no extension length field either)
    if (getOffset() + m_hello.compressionMethodsLen == expectedLength){
        // No extensions - ClientHello ends after compression methods
        
        // Defense-in-depth: Verify compression methods data actually exists in buffer
        // This prevents trusting a malicious compressionMethodsLen that claims more data than present
        if (!canReadLength(m_hello.compressionMethodsLen)) {
            m_status = Status::Failure;
            return m_status;
        }
        
        consumed = getConsumed() + m_hello.compressionMethodsLen;
        
        // Validate lengths (no extension length field in packet)
        if (!validateLengths(expectedLength, false)) {
            m_status = Status::Failure;
            return m_status;
        }
        
        m_status  = Status::Complete;
        return m_status;
    }

    // Skip all compression methods
    if (!incOffset(m_hello.compressionMethodsLen)){
        m_status = Status::IncompleteClientHelloBody;
        return m_status;
    }

    // Check if extensions length filed is within payload
    if (!canReadLength(extensionLengthFieldLen)){
        m_status = Status::IncompleteClientHelloBody;
        return m_status;
    }
    // Read extensions length field
    m_hello.extensionLen = getDataAtOffsetU16();

    // Validate that sum of all field lengths matches expected ClientHello length
    // Extension length field IS present in packet (even if extensionLen=0)
    if (!validateLengths(expectedLength, true)) {
        m_status = Status::Failure;
        return m_status;
    }

    // Skip extensions length field
    if (!incOffset(extensionLengthFieldLen)){
        return m_status;
    }

    // Save pointer to extensions
    m_hello.pExtensions = getDataPtrAtOffset();

    // Skip all extensions
    if (m_hello.extensionLen == 0) {
        consumed = getConsumed();
        m_status = Status::Complete;
        return m_status;
    }

    if (!incOffset(m_hello.extensionLen - 1)){
        m_status = Status::IncompleteExtensionSection;
        consumeAll();
        consumed = getConsumed();
        return m_status;
    }

    consumed = getConsumed();
    
    m_status  = Status::Complete;
    return m_status;
}

bool ClientHelloParser::validateLengths(uint32_t expectedLength, bool hasExtensionLengthField) const
{
    // Calculate sum of all field lengths
    uint32_t sumFieldsLength = versionLength                     // 2 bytes
                              + randomLength                      // 32 bytes
                              + sessionIdLenFiledLen              // 1 byte
                              + m_hello.sidLen                    // variable
                              + m_hello.suitesLen                 // variable (includes 2-byte length field)
                              + compressionsLengthFieldLen        // 1 byte
                              + m_hello.compressionMethodsLen;    // variable
    
    // Only add extension length field if it exists in the packet
    // (no extensions path = no extension length field per RFC 5246)
    if (hasExtensionLengthField) {
        sumFieldsLength += extensionLengthFieldLen;    // 2 bytes (always present if field exists)
        sumFieldsLength += m_hello.extensionLen;       // variable (may be 0)
    }
                               
    if (expectedLength != sumFieldsLength) {
        return false;
    }                
    
    return true;
}


IExtensionsParser::Status ExtensionsParser::parse(const uint8_t *const payload, uint16_t len, uint16_t &consumed) {

    m_status = Status::Failure;

    m_extensionType = 0;
    m_extensionLen = 0;
    m_extensionDataPtr = nullptr;

    if (!init(payload, len)) {
        return m_status;
    }
    m_status = Status::PendingTypeField;

    if (len == 0) {
        return m_status;
    }

    if (!canReadLength(1 )){ //
        return m_status;
    }
    m_status = Status::IncompleteExtensionTypeField;
    if (!canReadLength(m_extensionTypeFieldLen)){
        uint8_t  firstByte = getDataAtOffsetU8();
        m_extensionType = firstByte << 8;
        return m_status;
    }

    // Got the complete extension type field
    m_extensionType = getDataAtOffsetU16();

    m_status = Status::PendingLengthField;
    if (!incOffset(m_extensionTypeFieldLen)){
        return m_status;
    }

    m_status = Status::IncompleteExtensionLengthField;
    if (!canReadLength(m_extensionLengthFieldLen)){
        uint8_t  firstByte = getDataAtOffsetU8();
        m_extensionLen = firstByte << 8;
        return m_status;
    }

    // Got the complete extension Length Field
    m_extensionLen = getDataAtOffsetU16();
    const uint16_t extensionDataBytes =
        static_cast<uint16_t>(len - m_extensionTypeFieldLen - m_extensionLengthFieldLen);
    if(m_extensionLen > extensionDataBytes) {
        m_status = Status::Failure;
        return m_status; // extension length exceeds available data
    }

    m_status = Status::PendingExtensionData;

    if (0 == m_extensionLen){
        incOffset(m_extensionLengthFieldLen - 1);
        consumed = getConsumed();
        m_status = Status::Complete;
        return m_status;
    }

    if (!incOffset(m_extensionLengthFieldLen)){
        return m_status;
    }

    m_status = Status::IncompleteExtensionData;

    m_extensionDataPtr = getDataPtrAtOffset();

    if(isExtensionDataValid() == false) {
        m_status = Status::Failure;
        return m_status; // invalid extension data pointer
    }

    if (!incOffset(m_extensionLen - 1)){
        return m_status;
    }
    consumed = getConsumed();
    m_status = Status::Complete;
    return m_status;
}

bool ExtensionsParser::isExtensionDataValid() const
{
    uint32_t numOfBytesOfDataLengthField = getNumOfBytesOfDataLengthField();

    if (numOfBytesOfDataLengthField == 0) {
        // Extension doesn't have simple length field at byte 0
        // Check if it needs specific validation
        return validateComplexExtension();
    }
    
    // Check if extension has enough bytes for the length field
    if (m_extensionLen < numOfBytesOfDataLengthField) {
        return false;  // Malformed: extension too small for its type
    }

    uint16_t extensionDataLength{0};

    if(numOfBytesOfDataLengthField == 2)
    {
        extensionDataLength = (m_extensionDataPtr[0] << 8) | m_extensionDataPtr[1];
    }
    else if(numOfBytesOfDataLengthField == 1)
    {
        extensionDataLength = m_extensionDataPtr[0];
    }
    
    if(extensionDataLength + numOfBytesOfDataLengthField > m_extensionLen)
    {
        return false; // Malformed: inner length exceeds extension length
    }

    return true;
}

bool ExtensionsParser::validateComplexExtension() const
{
    switch (static_cast<ExtensionTypeE>(m_extensionType)) {
        case ExtensionTypeE::status_request:
            // RFC 6066: Byte 0 = status_type, then responder_id_list with 2-byte length
            if (m_extensionLen < 1) return false;
            {
                uint8_t status_type = m_extensionDataPtr[0];
                // Valid status types: 1 (ocsp), 2 (ocsp_multi)
                if (status_type != 1 && status_type != 2) return false;
                
                // If there's more data, validate responder_id_list length
                if (m_extensionLen >= 3) {
                    uint16_t responder_list_len = (m_extensionDataPtr[1] << 8) | m_extensionDataPtr[2];
                    if (responder_list_len + 3 > m_extensionLen) return false;
                }
            }
            return true;

        case ExtensionTypeE::status_request_v2:
            // Similar to status_request but with version 2 structure
            if (m_extensionLen < 3) return false;
            {
                uint8_t status_type = m_extensionDataPtr[2];
                if (status_type != 1 && status_type != 2) return false;
            }
            return true;

        case ExtensionTypeE::pre_shared_key:
            // RFC 8446: identities (2-byte length) then binders (2-byte length)
            // Too complex to validate fully here without knowing context
            // At minimum, check we have room for identities length
            if (m_extensionLen < 2) return false;
            {
                uint16_t identities_len = (m_extensionDataPtr[0] << 8) | m_extensionDataPtr[1];
                // Identities + binders length field (2 bytes minimum)
                if (identities_len + 4 > m_extensionLen) return false;
            }
            return true;

        case ExtensionTypeE::key_share:
            // RFC 8446: Different structure for ClientHello vs ServerHello
            // ClientHello: client_shares length (2 bytes) + shares
            // ServerHello: single KeyShareEntry (no length prefix)
            // For safety, just check minimum size
            if (m_extensionLen < 2) return false;
            return true;

        // Extensions without inner length fields - no validation needed
        case ExtensionTypeE::session_ticket:
        case ExtensionTypeE::heartbeat:
        case ExtensionTypeE::extended_master_secret:
        case ExtensionTypeE::padding:
        default:
            // No specific validation for these extensions
            return true;
    }
}

uint32_t ExtensionsParser::getNumOfBytesOfDataLengthField() const
{
    switch (static_cast<ExtensionTypeE>(m_extensionType)) {
        /* Extensions with 2-byte length field AT BYTE 0 */
        case ExtensionTypeE::server_name:                               // RFC 6066: ServerNameList length
        case ExtensionTypeE::supported_groups:                          // RFC 8422: NamedGroupList length
        case ExtensionTypeE::signature_algorithms:                      // RFC 5246: supported_signature_algorithms length
        case ExtensionTypeE::application_layer_protocol_negotiation:    // RFC 7301: ProtocolNameList length
        case ExtensionTypeE::signature_algorithms_cert:                 // RFC 8446: supported_signature_algorithms length
        case ExtensionTypeE::certificate_authorities:                   // RFC 8446: authorities length
        case ExtensionTypeE::cookie:                                    // RFC 8446: cookie length
            return 2;

        /* Extensions with 1-byte length field AT BYTE 0 */
        case ExtensionTypeE::ec_point_formats:                          // RFC 4492: ECPointFormatList length
        case ExtensionTypeE::supported_versions:                        // RFC 8446: versions length (ClientHello)
        case ExtensionTypeE::psk_key_exchange_modes:                    // RFC 8446: ke_modes length
        case ExtensionTypeE::compress_certificate:                      // RFC 8879: algorithms length
            return 1;

        /* Extensions with complex structure - validated in validateComplexExtension() */
        case ExtensionTypeE::status_request:        // Has status_type (1 byte) BEFORE length fields
        case ExtensionTypeE::status_request_v2:     // Has status_type (1 byte) BEFORE length fields
        case ExtensionTypeE::pre_shared_key:        // Multiple length fields (identities + binders)
        case ExtensionTypeE::key_share:             // Different structure in ClientHello vs ServerHello
        
        /* Extensions without inner length fields */
        default:
            return 0;
    }
}

ParsingStatus SniParser::parse(const uint8_t *const payload, uint16_t len, uint16_t &consumed) {

	m_status = ParsingStatus::Failure;

    if (!init(payload, len)) {
        return m_status;
    }
    if (!canReadLength(sniListFieldLength)){
        return m_status;
    }
    m_SniListLength = getDataAtOffsetU16();

    // RFC 6066 + DoS hardening: SNI list must not exceed 64 bytes
    if (m_SniListLength > max_sni_list_length) {
        return m_status;
    }

    if (!incOffset(sniListFieldLength)){
        return m_status;
    }

    uint16_t  currSniLen;
    m_numOfSnis = 0;
    while (canReadLength(1)) {
        if (m_numOfSnis < max_snis_supported) {
            m_sniTypes[m_numOfSnis] = getDataAtOffsetU8();
        }
        if (!incOffset(sniTypeFieldLength)){
            return m_status;
        }
        if (!canReadLength(sniLengthFieldLength)){
            return m_status;
        }
        currSniLen = getDataAtOffsetU16();

        // RFC 6066: SNI name must be at least 1 byte and at most 255 bytes.
        // Zero-length names are malformed and should be rejected.
        if (currSniLen == 0 || currSniLen > TLSFlow::max_sni_length) {
            return m_status;
        }
        if (m_numOfSnis < max_snis_supported) {
            m_sniLengths[m_numOfSnis] = currSniLen;
        }
        if (!incOffset(sniLengthFieldLength)){
            return m_status;
        }
        if (m_numOfSnis < max_snis_supported) {
            m_sniPointers[m_numOfSnis] = getDataPtrAtOffset();
        }
        if (!incOffset(currSniLen - 1)){
            return m_status;
        }
        // Ended parsing of SNI record, count it
        m_numOfSnis++;
        // Check if we have parsed the complete list of SNIs
        if (getConsumed() == m_SniListLength + sniListFieldLength) {
            m_status = ParsingStatus::Complete;
            break;
        }
    	if (getConsumed() > m_SniListLength + sniListFieldLength){
    		// Consumed somehow more than the declared size, could be malformed extension on top level where the input len function argument is too long
    		return m_status;
    	}
    	// prepare to next SNI
    	if (!incOffset(1)) {
    		return m_status;
    	}
    }

    return m_status;
}

ParsingStatus TlsAlertParser::parse(const uint8_t *const payload, uint16_t len, uint16_t &consumed)
{
    m_status = ParsingStatus::Failure;
    consumed = 0;

    if (!init(payload, len))
    {
        return m_status;
    }

    m_alertLevel = getDataAtOffsetU8();
    incOffset(1);
    m_alertDescription = getDataAtOffsetU8();
    consumed = getConsumed();
    m_status = ParsingStatus::Complete;
    return m_status;
}

void printOutput(TERMG_ID termId, std::stringstream& output)
{
    std::string line;
    while (std::getline(output, line))

bool parseUnsigned(const char* text, uint32_t& value)
{
    if (text == nullptr || *text == '\0' || *text == '-')
    {
        return false;
    }

    errno = 0;
    char* end = nullptr;
    const unsigned long parsed = std::strtoul(text, &end, 10);
    if (errno != 0 || end == text || *end != '\0' || parsed > UINT32_MAX)
    {
        return false;
    }
    value = static_cast<uint32_t>(parsed);
    return true;
}

bool parseSessionIndex(const char* text, int32_t& value)
{
    if (text == nullptr || *text == '\0')
    {
        return false;
    }

    errno = 0;
    char* end = nullptr;
    const long parsed = std::strtol(text, &end, 10);
    if (errno != 0 || end == text || *end != '\0' || parsed < INT32_MIN || parsed > INT32_MAX)
    {
        return false;
    }
    value = static_cast<int32_t>(parsed);
    return true;
}

bool parseCore(const char* text, uint32_t& coreId)
{
    if (std::strcmp(text, "all") == 0)
    {
        coreId = 0;
        return true;
    }

    return parseUnsigned(text, coreId) && coreId > 0;
}

TERMG_ERROR TLSFlow::NSDiagnostics::Controller::printEventsStatistics(TERMG_ID id, UINT_32 argc, char *argv[]) {
    termg_printf(id, "\nWeb DDoS Aggregated Counter Events:\n\n");
    uint64_t val;

    static constexpr char eventNames[to_underlying(NSDiagnostics::Common::DiagnosticsEvents::MaxEventId)][50] = {
        "Segmented State allocations                   ",
        "Segmented State frees                         ",
        "Segmented State Reused                        ",
        "Segmented State allocation failures           ",
        "Unexpected tcp seq during segmented process   ",
        "Unexpected length during segmented process    ",
        "Unsegmented ClientHello completed             ",
        "Two-segment ClientHello completed             ",
        "Three-or-more-segment ClientHello completed  ",
        "First Segment Prcocess success                ",
        "Extensions Parser Failures                    ",
        "General Parser failures                       ",
        "Unsupported version                           ",
        "Unsupported extensions exceed buffer          ",
        "Unsupported third segment required            ",
        "Unsupported record-level fragmentation        ",
        "Segmented state wrong sequence                ",
        "Unsupported early ClientHello body            ",
        "Continuation gap                              ",
        "ClientHello continuation started              ",
        "ClientHello continuation gap                  ",
        "Opaque record continuation started            ",
        "Opaque record continuation gap                ",
        "Continuation full retransmission              ",
        "Continuation partial overlap                  ",
        "Segment buffer full                           ",
        };

    bool bGet = false, bReset = false;

    if (argc == 1) {
        bGet = true;
    }
    else if (argc == 2) {
        std::string command = argv[1];
        if (command == "get") {
            bGet = true;
        }else if (command == "reset"){

            bReset = true;
        }
        else {
            printDiagnosticsStatisticsHelp(id);
            return TERMG_error_okey_E;
        }
    }


    if(bGet) {
        for(uint8_t event = 0; event < to_underlying(NSDiagnostics::Common::DiagnosticsEvents::MaxEventId); ++event) {
            val = TLSFlow::Context::Controller::Application::getInstance().getDiagnosticsApi()->getDiagnosticsCounter(static_cast<NSDiagnostics::Common::DiagnosticsEvents>(event));
            termg_printf(id, "%s : %llu\n", eventNames[event], val);
        }
        termg_printf(id, "\n");
    }
    else if (bReset) {
        for(uint8_t event = 0; event < to_underlying(NSDiagnostics::Common::DiagnosticsEvents::MaxEventId); ++event) {
            TLSFlow::Context::Controller::Application::getInstance().getDiagnosticsApi()->resetDiagnosticsCounter(static_cast<NSDiagnostics::Common::DiagnosticsEvents>(event));
        }
        termg_printf(id, "Ressetting Event Counters - Done\n");
    }
}

TERMG_ERROR TLSFlow::NSDiagnostics::Controller::printSessionExtensionTableSize(TERMG_ID termId, UINT_32 argc, char* argv[])
{
    if (argc != 2 || std::strcmp(argv[1], "help") == 0)
    {
        printUsage(termId);
        return argc == 2 ? TERMG_error_okey_E : TERMG_error_syntax_E;
    }

    uint32_t coreId = 0;
    if (!parseCore(argv[1], coreId))
    {
        termg_printf(termId, "Error: invalid engine index.\n");
        return TERMG_error_unknown_parameter_E;
    }

    auto manager = Application::getInstance().getSessionExtensionInspector();
    if (manager == nullptr || !manager->requestSize(coreId))
    {
        termg_printf(termId, "Error: failed to fetch TLS session extension table size.\n");
        return TERMG_error_error_E;
    }

    std::stringstream output;
    manager->fillSizeResultStream(&output);
    printOutput(termId, output);
    return TERMG_error_okey_E;
}

TERMG_ERROR TLSFlow::NSDiagnostics::Controller::searchSessionExtensionTable(TERMG_ID termId, UINT_32 argc, char* argv[])
{
    if (argc != 3 || std::strcmp(argv[1], "help") == 0)
    {
        printUsage(termId);
        return argc == 2 ? TERMG_error_okey_E : TERMG_error_syntax_E;
    }

    uint32_t coreId = 0;
    int32_t sessionIndex = 0;
    if (!parseCore(argv[1], coreId) || !parseSessionIndex(argv[2], sessionIndex))
    {
        termg_printf(termId, "Error: invalid engine index or session index.\n");
        return TERMG_error_unknown_parameter_E;
    }

    auto manager = Application::getInstance().getSessionExtensionInspector();
    if (manager == nullptr || !manager->requestEntries(coreId, false, sessionIndex, 1))
    {
        termg_printf(termId, "Error: failed to search TLS session extension table.\n");
        return TERMG_error_error_E;
    }

    std::stringstream output;
    if (manager->hasEntryResults())
    {
        manager->fillEntryResultStream(&output);
    }
    else
    {
        output << "No TLS session extension found for session index " << sessionIndex << ".\n";
    }
    printOutput(termId, output);
    return TERMG_error_okey_E;
}

TERMG_ERROR TLSFlow::NSDiagnostics::Controller::printSessionExtensionTable(TERMG_ID termId, UINT_32 argc, char* argv[])
{
    if (argc != 4 || std::strcmp(argv[1], "help") == 0)
    {
        printUsage(termId);
        return argc == 2 ? TERMG_error_okey_E : TERMG_error_syntax_E;
    }

    uint32_t coreId = 0;
    uint32_t maxEntries = 0;
    bool allSessions = false;
    int32_t sessionIndex = 0;
    if (!parseCore(argv[1], coreId) || !parseUnsigned(argv[3], maxEntries) || maxEntries == 0)
    {
        termg_printf(termId, "Error: invalid engine index or maximum entries.\n");
        return TERMG_error_unknown_parameter_E;
    }

    if (std::strcmp(argv[2], "all") == 0)
    {
        allSessions = true;
    }
    else if (!parseSessionIndex(argv[2], sessionIndex))
    {
        termg_printf(termId, "Error: invalid session index.\n");
        return TERMG_error_unknown_parameter_E;
    }

    auto manager = Application::getInstance().getSessionExtensionInspector();
    if (manager == nullptr || !manager->requestEntries(coreId, allSessions, sessionIndex, maxEntries))
    {
        termg_printf(termId, "Error: failed to print TLS session extension table.\n");
        return TERMG_error_error_E;
    }

    std::stringstream output;
    if (manager->hasEntryResults())
    {
        manager->fillEntryResultStream(&output);
    }
    else
    {
        output << "TLS session extension table is empty.\n";
    }
    printOutput(termId, output);
    return TERMG_error_okey_E;
}
